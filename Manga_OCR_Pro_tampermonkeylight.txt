    // Enhanced Cache with TTL and Size Limits
    class ImageDataCache {
        constructor(maxSize = APP_CONFIG.CACHE.MAX_IMAGES) {
            this.cache = new WeakMap();
            this.keys = new Set();
            this.maxSize = maxSize;
            this.creationTimes = new WeakMap();
        }

        set(key, value) {
            if (this.keys.size >= this.maxSize) {
                const firstKey = this.keys.values().next().value;
                this.cache.delete(firstKey);
                this.creationTimes.delete(firstKey);
                this.keys.delete(firstKey);
            }
            this.cache.set(key, value);
            this.creationTimes.set(key, Date.now());
            this.keys.add(key);
        }

        get(key) {
            return this.cache.get(key);
        }

        cleanup(ttl = APP_CONFIG.CACHE.EMBEDDING_TTL) {
            const now = Date.now();
            for (const key of this.keys) {
                const creationTime = this.creationTimes.get(key);
                if (creationTime && (now - creationTime) > ttl) {
                    this.cache.delete(key);
                    this.creationTimes.delete(key);
                    this.keys.delete(key);
                }
            }
        }
    }

    // Enhanced MAGI API request with retry logic
    async function magiRequestWithRetry(url, data, maxRetries = APP_CONFIG.PROCESSING.MAX_RETRIES) {
        for (let attempt = 1; attempt <= maxRetries; attempt++) {
            try {
                const result = await magiRequest(url, data);
                return result;
            } catch (error) {
                if (attempt === maxRetries) {
                    throw new Error(`MAGI request failed after ${maxRetries} attempts: ${error.message}`);
                }

                console.warn(`MAGI request failed, retrying... (${attempt}/${maxRetries})`);

                // Exponential backoff
                const backoffTime = 1000 * Math.pow(2, attempt - 1);
                await new Promise(resolve => setTimeout(resolve, backoffTime));
            }
        }
    }

    // Enhanced overlay positioning system
    class OverlayPositioningSystem {
        constructor() {
            this.overlays = new WeakMap();
            this.observedContainers = new WeakSet();
            this.resizeObserver = null;
            this.intersectionObserver = null;
            this.mutationObserver = null;
            this.scrollHandler = null;
            this.updateTimeouts = new WeakMap();
            this.globalUpdateTimeout = null;
        }

        init() {
            this.setupObservers();
            this.setupScrollHandler();
        }

        setupObservers() {
            // ResizeObserver for container and image size changes
            this.resizeObserver = new ResizeObserver(entries => {
                entries.forEach(entry => {
                    const container = entry.target;
                    this.updateOverlaysInContainer(container);
                });
            });

            // IntersectionObserver for visibility changes
            this.intersectionObserver = new IntersectionObserver(entries => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        const container = entry.target;
                        this.updateOverlaysInContainer(container);
                    }
                });
            }, {
                threshold: 0.1,
                rootMargin: '50px' // Update when near viewport
            });

            // MutationObserver for DOM changes
            this.mutationObserver = new MutationObserver(mutations => {
                let needsUpdate = false;

                mutations.forEach(mutation => {
                    if (mutation.type === 'attributes') {
                        const target = mutation.target;
                        if (this.overlays.has(target)) {
                            this.debouncedUpdate(target);
                        }
                    } else if (mutation.addedNodes.length > 0 || mutation.removedNodes.length > 0) {
                        needsUpdate = true;
                    }
                });

                if (needsUpdate) {
                    this.checkForLayoutChanges();
                }
            });
        }

        setupScrollHandler() {
            this.scrollHandler = () => {
                this.debouncedGlobalUpdate();
            };
            window.addEventListener('scroll', this.scrollHandler, { passive: true });
        }

        monitorOverlay(overlay, container, img, bbox, lang) {
            this.overlays.set(overlay, { container, img, bbox, lang });

            // Start observing the container if not already being observed
            if (!this.observedContainers.has(container)) {
                this.resizeObserver.observe(container);
                this.intersectionObserver.observe(container);
                this.mutationObserver.observe(container, {
                    attributes: true,
                    childList: true,
                    subtree: true,
                    attributeFilter: ['style', 'class', 'id']
                });
                this.observedContainers.add(container);
            }

            // Initial positioning
            this.updateOverlayPosition(overlay);
        }

        updateOverlaysInContainer(container) {
            const overlays = container.querySelectorAll('.ocr-overlay');
            overlays.forEach(overlay => {
                const data = this.overlays.get(overlay);
                if (data && data.container === container) {
                    this.updateOverlayPosition(overlay);
                }
            });
        }

        updateOverlayPosition(overlay) {
            const data = this.overlays.get(overlay);
            if (!data) return;

            const { container, img, bbox } = data;

            // Validate container and image still exist
            if (!document.contains(container) || !document.contains(img)) {
                this.overlays.delete(overlay);
                overlay.remove();
                return;
            }

            try {
                // Get current image dimensions and position
                const imgRect = img.getBoundingClientRect();
                const containerRect = container.getBoundingClientRect();

                // Validate dimensions
                if (imgRect.width === 0 || imgRect.height === 0 ||
                    containerRect.width === 0 || containerRect.height === 0) {
                    return;
                }

                // Calculate scaling factors
                const scaleX = imgRect.width / Math.max(img.naturalWidth, 1);
                const scaleY = imgRect.height / Math.max(img.naturalHeight, 1);

                // Apply scaling to original bbox
                const scaledLeft = bbox[0] * imgRect.width;
                const scaledTop = bbox[1] * imgRect.height;
                const scaledWidth = Math.max((bbox[2] - bbox[0]) * imgRect.width, 10); // Minimum width
                const scaledHeight = Math.max((bbox[3] - bbox[1]) * imgRect.height, 10); // Minimum height

                // Calculate final position relative to container
                const containerX = containerRect.left - imgRect.left;
                const containerY = containerRect.top - imgRect.top;

                const finalLeft = Math.max(0, scaledLeft + containerX);
                const finalTop = Math.max(0, scaledTop + containerY);
                const finalWidth = Math.min(scaledWidth, containerRect.width - finalLeft);
                const finalHeight = Math.min(scaledHeight, containerRect.height - finalTop);

                // Validate final dimensions
                if (finalWidth <= 0 || finalHeight <= 0) {
                    overlay.style.display = 'none';
                    return;
                }

                // Apply positioning
                overlay.style.left = `${finalLeft}px`;
                overlay.style.top = `${finalTop}px`;
                overlay.style.width = `${finalWidth}px`;
                overlay.style.minHeight = `${finalHeight}px`;
                overlay.style.display = 'block';

                // Check for collisions with page elements
                this.avoidElementCollisions(overlay, container);
            } catch (error) {
                console.warn('Error updating overlay position:', error);
            }
        }

        avoidElementCollisions(overlay, container) {
            try {
                const overlayRect = overlay.getBoundingClientRect();

                // Get all potentially colliding elements (limit to reasonable scope)
                const pageElements = document.elementsFromPoint(
                    overlayRect.left + overlayRect.width / 2,
                    overlayRect.top + overlayRect.height / 2
                );

                let hasCollision = false;

                for (const element of pageElements) {
                    if (element === overlay || element === container ||
                        element.contains(overlay) || overlay.contains(element)) {
                        continue;
                    }

                    const elementRect = element.getBoundingClientRect();

                    if (this.elementsOverlap(overlayRect, elementRect)) {
                        this.adjustOverlayForCollision(overlay, elementRect, container);
                        hasCollision = true;
                        break;
                    }
                }

                // Reset opacity if no collisions
                if (!hasCollision) {
                    overlay.style.opacity = '';
                }
            } catch (error) {
                console.warn('Error in collision avoidance:', error);
            }
        }

        elementsOverlap(rect1, rect2) {
            return !(
                rect1.right < rect2.left ||
                rect1.left > rect2.right ||
                rect1.bottom < rect2.top ||
                rect1.top > rect2.bottom
            );
        }

        adjustOverlayForCollision(overlay, obstacleRect, container) {
            const overlayRect = overlay.getBoundingClientRect();
            const containerRect = container.getBoundingClientRect();

            // Try different positioning strategies
            const strategies = [
                // Right side of obstacle
                { x: obstacleRect.right + 5, y: overlayRect.top },
                // Below obstacle
                { x: overlayRect.left, y: obstacleRect.bottom + 5 },
                // Left side of obstacle
                { x: obstacleRect.left - overlayRect.width - 5, y: overlayRect.top },
                // Above obstacle
                { x: overlayRect.left, y: obstacleRect.top - overlayRect.height - 5 }
            ];

            // Find the first valid position
            for (const position of strategies) {
                if (this.isPositionValid(position, overlayRect, containerRect)) {
                    // Convert to relative coordinates within container
                    const relativeX = position.x - containerRect.left;
                    const relativeY = position.y - containerRect.top;

                    overlay.style.left = `${relativeX}px`;
                    overlay.style.top = `${relativeY}px`;
                    overlay.style.opacity = '0.9';
                    return;
                }
            }

            // If no valid position found, reduce opacity
            overlay.style.opacity = '0.8';
        }

        isPositionValid(position, overlayRect, containerRect) {
            const testRect = {
                left: position.x,
                top: position.y,
                right: position.x + overlayRect.width,
                bottom: position.y + overlayRect.height
            };

            // Check if within container bounds
            const withinContainer = (
                testRect.left >= containerRect.left &&
                testRect.top >= containerRect.top &&
                testRect.right <= containerRect.right &&
                testRect.bottom <= containerRect.bottom
            );

            // Check if within viewport
            const withinViewport = (
                testRect.left >= 0 &&
                testRect.top >= 0 &&
                testRect.right <= window.innerWidth &&
                testRect.bottom <= window.innerHeight
            );

            return withinContainer && withinViewport;
        }

        debouncedUpdate(overlay) {
            clearTimeout(this.updateTimeouts.get(overlay));
            const timeout = setTimeout(() => {
                this.updateOverlayPosition(overlay);
            }, 100);
            this.updateTimeouts.set(overlay, timeout);
        }

        debouncedGlobalUpdate() {
            clearTimeout(this.globalUpdateTimeout);
            this.globalUpdateTimeout = setTimeout(() => {
                this.updateAllOverlayPositions();
            }, 150);
        }

        updateAllOverlayPositions() {
            document.querySelectorAll('.ocr-overlay').forEach(overlay => {
                this.updateOverlayPosition(overlay);
            });
        }

        checkForLayoutChanges() {
            this.debouncedGlobalUpdate();
        }

        destroy() {
            if (this.resizeObserver) this.resizeObserver.disconnect();
            if (this.intersectionObserver) this.intersectionObserver.disconnect();
            if (this.mutationObserver) this.mutationObserver.disconnect();
            if (this.scrollHandler) window.removeEventListener('scroll', this.scrollHandler);

            // Clear all timeouts
            for (const timeout of Object.values(this.updateTimeouts)) {
                clearTimeout(timeout);
            }
            clearTimeout(this.globalUpdateTimeout);
        }
    }

    // MAGI API request helper
    function magiRequest(url, data) {
        return new Promise((resolve, reject) => {
            GM_xmlhttpRequest({
                method: 'POST',
                url: url,
                data: JSON.stringify(data),
                headers: {'Content-Type': 'application/json'},
                timeout: APP_CONFIG.MAGI_API.TIMEOUT,
                onload: (res) => {
                    if (res.status >= 200 && res.status < 300) {
                        try {
                            resolve(JSON.parse(res.responseText));
                        } catch (e) {
                            reject(new Error(`Failed to parse response: ${e.message}`));
                        }
                    } else {
                        reject(new Error(`HTTP ${res.status}: ${res.statusText}`));
                    }
                },
                onerror: (err) => reject(new Error(`Network error: ${err.statusText}`)),
                ontimeout: () => reject(new Error('Request timeout'))
            });
        });
    }

    // Enhanced image validation and base64 conversion
    async function getImageBase64(img) {
        // Validate image
        if (!img || !img.src) {
            throw new Error('Invalid image element');
        }

        // Use proxy only for cross-origin images
        if (img.src.startsWith('data:') ||
            img.src.startsWith('blob:') ||
            new URL(img.src, window.location.href).origin === location.origin) {

            return new Promise((resolve, reject) => {
                if (!img.complete || img.naturalHeight === 0) {
                    reject(new Error('Image not loaded'));
                    return;
                }

                try {
                    const canvas = document.createElement('canvas');
                    const ctx = canvas.getContext('2d');
                    canvas.width = img.naturalWidth;
                    canvas.height = img.naturalHeight;
                    ctx.drawImage(img, 0, 0);
                    resolve(canvas.toDataURL('image/png'));
                } catch (error) {
                    reject(new Error(`Canvas conversion failed: ${error.message}`));
                }
            });
        }

        // Use proxy for cross-origin images
        return new Promise((resolve, reject) => {
            GM_xmlhttpRequest({
                method: 'POST',
                url: APP_CONFIG.CORS_PROXY,
                data: JSON.stringify({ imageUrl: img.src }),
                headers: {'Content-Type': 'application/json'},
                timeout: 15000,
                onload: (res) => {
                    try {
                        const data = JSON.parse(res.responseText);
                        if (data.base64) {
                            resolve(data.base64);
                        } else {
                            reject(new Error('No base64 data in response'));
                        }
                    } catch (e) {
                        reject(new Error(`Proxy response parsing failed: ${e.message}`));
                    }
                },
                onerror: (err) => {
                    reject(new Error(`CORS proxy failed: ${err.statusText}`));
                },
                ontimeout: () => {
                    reject(new Error('CORS proxy timeout'));
                }
            });
        });
    }

    // Enhanced Character Mapper with persistence
    class CharacterMapper {
        constructor() {
            this.clusterToSpeaker = {};
            this.nextSpeakerId = 1;
            this.storedEmbeds = {};
            this.loadFromStorage();
        }

        loadFromStorage() {
            try {
                const stored = GM_getValue('character_mapper_data');
                if (stored) {
                    const data = JSON.parse(stored);
                    this.clusterToSpeaker = data.clusterToSpeaker || {};
                    this.nextSpeakerId = data.nextSpeakerId || 1;
                    this.storedEmbeds = data.storedEmbeds || {};
                }
            } catch (error) {
                console.warn('Failed to load character mapper from storage:', error);
            }
        }

        saveToStorage() {
            try {
                GM_setValue('character_mapper_data', JSON.stringify({
                    clusterToSpeaker: this.clusterToSpeaker,
                    nextSpeakerId: this.nextSpeakerId,
                    storedEmbeds: this.storedEmbeds
                }));
            } catch (error) {
                console.warn('Failed to save character mapper to storage:', error);
            }
        }

        getSpeaker(embeddings) {
            const globalId = this.findBestMatch(embeddings);

            if (!this.clusterToSpeaker[globalId]) {
                this.clusterToSpeaker[globalId] = `CHAR${this.nextSpeakerId++}`;
                this.saveToStorage();
            }
            return this.clusterToSpeaker[globalId];
        }

        findBestMatch(embedding) {
            // Validation
            if (!Array.isArray(embedding) || embedding.length === 0) {
                return this.createNewCharacter(embedding);
            }

            // First character storage
            if (Object.keys(this.storedEmbeds).length === 0) {
                const newId = 1;
                this.storedEmbeds[newId] = embedding;
                this.saveToStorage();
                return newId;
            }

            // Compare to stored embeddings
            let bestId = null;
            let bestSimilarity = 0;

            for (const [id, storedEmbeds] of Object.entries(this.storedEmbeds)) {
                const similarity = this.cosineSimilarity(embedding, storedEmbeds);
                if (similarity > bestSimilarity && similarity > APP_CONFIG.SIMILARITY_THRESHOLD) {
                    bestSimilarity = similarity;
                    bestId = parseInt(id);
                }
            }

            if (bestId !== null) {
                return bestId;
            }

            // New character
            return this.createNewCharacter(embedding);
        }

        createNewCharacter(embedding) {
            const newId = Object.keys(this.storedEmbeds).length + 1;
            this.storedEmbeds[newId] = embedding;
            this.saveToStorage();
            return newId;
        }

        cosineSimilarity(embedding, storedEmbeds) {
            // Validation
            if (!Array.isArray(embedding) || !Array.isArray(storedEmbeds)) {
                return 0;
            }
            if (embedding.length !== storedEmbeds.length) {
                return 0;
            }

            // Calculate dot product
            let dot = 0;
            for (let i = 0; i < embedding.length; i++) {
                dot += embedding[i] * storedEmbeds[i];
            }

            // Calculate magnitudes
            let magA = 0;
            let magB = 0;
            for (let i = 0; i < embedding.length; i++) {
                magA += embedding[i] * embedding[i];
                magB += storedEmbeds[i] * storedEmbeds[i];
            }
            magA = Math.sqrt(magA);
            magB = Math.sqrt(magB);

            // Avoid division by zero
            if (magA === 0 || magB === 0) {
                return 0;
            }

            return dot / (magA * magB);
        }

        clear() {
            this.clusterToSpeaker = {};
            this.nextSpeakerId = 1;
            this.storedEmbeds = {};
            GM_deleteValue('character_mapper_data');
        }
    }

    // Enhanced language detection
    function detectLanguageWithConfidence(text) {
        if (!text || text.length < 3) return { lang: 'eng', confidence: 0 };

        const patterns = {
            'jpn': [/[ぁ-ん]/, /[ァ-ン]/, /[一-龯]/], // Hiragana, Katakana, Kanji
            'kor': [/[\uAC00-\uD7AF]/], // Hangul
            'chi_sim': [/[\u4E00-\u9FFF]/], // Chinese
            'ru': [/[А-Яа-яЁё]/], // Cyrillic
            'eng': [/[a-zA-Z]/] // English
        };

        let bestLang = 'eng';
        let bestScore = 0;

        for (const [lang, langPatterns] of Object.entries(patterns)) {
            let score = 0;
            for (const pattern of langPatterns) {
                const matches = text.match(pattern);
                if (matches) score += matches.length;
            }

            if (score > bestScore) {
                bestScore = score;
                bestLang = lang;
            }
        }

        const confidence = Math.min(bestScore / text.length, 1);
        return { lang: bestLang, confidence };
    }

    // Text processing utilities
    function cleanText(text, lang) {
        if (typeof text !== 'string') return '';

        // Common punctuation and symbols
        const basePattern = 'A-Za-z0-9\\s.,!?\'"\\-—「」『』（）«»„“”';

        // Language-specific Unicode ranges
        let langRanges = '';

        switch (lang) {
            case 'ru': // Cyrillic
                langRanges = '\\u0400-\\u04FF';
                break;
            case 'chi_sim': // Chinese
                langRanges = '\\u4E00-\\u9FFF';
                break;
            case 'jpn': // Japanese
                langRanges = [
                    '\\u3040-\\u309F', // Hiragana
                    '\\u30A0-\\u30FF', // Katakana
                    '\\u31F0-\\u31FF', // Katakana Extensions
                    '\\u4E00-\\u9FFF', // Kanji
                    '\\uFF66-\\uFF9D'  // Half-width Katakana
                ].join('');
                break;
            case 'kor': // Korean
                langRanges = '\\uAC00-\\uD7AF';
                break;
            default:
                langRanges = '';
        }

        const whitelistRegex = new RegExp(`[^${basePattern}${langRanges}]`, 'gu');
        const multiSpaceRegex = /\s+/g;

        return text
            .replace(whitelistRegex, ' ')
            .replace(multiSpaceRegex, ' ')
            .trim();
    }

    function formatText(text) {
        if (!text) return '';
        const trimmed = text.trim();
        if (trimmed.length === 0) return '';
        return trimmed.charAt(0).toUpperCase() + trimmed.slice(1);
    }

    // Text-character association
    function associateTextToCharacters(texts, characters, panels, readingOrder = "manga") {
        const associations = [];

        panels.forEach(panel => {
            const panelBbox = panel.bbox;

            // Find characters in panel
            const panelChars = characters.filter(char =>
                isInside(char.bbox, panelBbox)
            );

            // Find texts in panel
            const panelTexts = texts.filter(text =>
                isInside(text.bbox, panelBbox)
            );

            // Associate each text to nearest character
            panelTexts.forEach(text => {
                let minDist = Infinity;
                let closestChar = null;

                panelChars.forEach(char => {
                    let dist = bboxDistance(text.bbox, char.bbox);

                    // Manga reading order preference (right side)
                    if (readingOrder === "manga") {
                        const textCenter = (text.bbox[0] + text.bbox[2]) / 2;
                        const charCenter = (char.bbox[0] + char.bbox[2]) / 2;
                        if (textCenter > charCenter) dist *= 0.7;
                    }

                    if (dist < minDist) {
                        minDist = dist;
                        closestChar = char;
                    }
                });

                if (closestChar) {
                    associations.push({
                        textIndex: texts.indexOf(text),
                        charIndex: characters.indexOf(closestChar)
                    });
                }
            });
        });

        return associations;
    }

    function isInside(inner, outer) {
        return (
            inner[0] >= outer[0] &&
            inner[1] >= outer[1] &&
            inner[2] <= outer[2] &&
            inner[3] <= outer[3]
        );
    }

    function bboxDistance(a, b) {
        const aCenter = [(a[0] + a[2]) / 2, (a[1] + a[3]) / 2];
        const bCenter = [(b[0] + b[2]) / 2, (b[1] + b[3]) / 2];
        return Math.sqrt(
            Math.pow(aCenter[0] - bCenter[0], 2) +
            Math.pow(aCenter[1] - bCenter[1], 2)
        );
    }

    // MAGI Processing Pipeline
    async function magiProcessImage(img, lang, readingOrder) {
        try {
            const detection = await magiBackend.detectComponents(img);
            const panels = detection.panels || [];
            const texts = detection.bubbles || [];
            const characters = detection.characters || [];

            if (!Array.isArray(characters) || characters.length === 0) {
                return {
                    texts,
                    characters,
                    associations: [],
                    ocrResults: [],
                    speakerMap: {}
                };
            }

            if (!Array.isArray(texts) || texts.length === 0) {
                return {
                    texts: [],
                    characters,
                    associations: [],
                    ocrResults: [],
                    speakerMap: {}
                };
            }

            const charBboxes = characters.map(c => c.bbox);
            const embeddings = await magiBackend.getCharacterEmbeddings(img, charBboxes);
            const charEmbeddings = embeddings.char_embeddings;

            const associations = associateTextToCharacters(texts, characters, panels, readingOrder);
            const textBboxes = texts.map(t => t.bbox);
            const ocrResults = await magiBackend.ocrTextRegions(img, textBboxes, lang);

            const mapper = getCharacterMapper();
            const speakerMap = {};
            for (let i = 0; i < characters.length; i++) {
                speakerMap[i] = mapper.getSpeaker(charEmbeddings[i]);
            }

            return {
                texts,
                characters,
                associations,
                ocrResults: ocrResults.results || [],
                speakerMap
            };
        } catch (error) {
            console.error('MAGI processing failed:', error);
            throw error;
        }
    }

    // Enhanced Image Processor with concurrency control
    class ImageProcessor {
        constructor(app) {
            this.app = app;
            this.pendingProcess = new Map();
            this.processingQueue = new Set();
            this.currentlyProcessing = new Map();
            this.maxConcurrent = APP_CONFIG.PROCESSING.MAX_CONCURRENT_PROCESSING;
        }

        async scheduleProcessing(img, priority = 'normal') {
            // Cancel existing pending processing for this image
            if (this.pendingProcess.has(img)) {
                clearTimeout(this.pendingProcess.get(img));
            }

            const timeoutId = setTimeout(async () => {
                this.pendingProcess.delete(img);
                await this.processImage(img);
            }, priority === 'high' ? 100 : APP_CONFIG.PROCESSING.DEBOUNCE_MS);

            this.pendingProcess.set(img, timeoutId);
        }

        async processImage(img) {
            if (!this.app.active ||
                this.currentlyProcessing.has(img.src) ||
                !img.complete ||
                img.naturalHeight === 0) {
                return;
            }

            // Check if already processed
            if (img.hasAttribute('data-magi-processed')) {
                return;
            }

            // Check backend health
            if (!connectionMonitor.isBackendHealthy) {
                console.warn('Skipping image processing - backend unhealthy');
                return;
            }

            // Limit concurrent processing
            if (this.currentlyProcessing.size >= this.maxConcurrent) {
                // Wait for a slot to free up
                await new Promise(resolve => setTimeout(resolve, 1000));
                return this.scheduleProcessing(img, 'high');
            }

            const processingPromise = this.processImageInternal(img);
            this.currentlyProcessing.set(img.src, processingPromise);

            try {
                await processingPromise;
            } finally {
                this.currentlyProcessing.delete(img.src);
            }
        }

        async processImageInternal(img) {
            try {
                // Create container if needed
                let container = img.parentElement;
                const isNewContainer = !container.classList.contains('magi-container');

                if (isNewContainer) {
                    container = document.createElement('div');
                    container.className = 'magi-container';
                    container.style.position = 'relative';
                    container.style.display = 'inline-block';
                    img.parentNode.insertBefore(container, img);
                    container.appendChild(img);
                }

                // Remove existing overlays for this image
                container.querySelectorAll('.ocr-overlay').forEach(el => el.remove());

                img.setAttribute('data-magi-processed', 'true');

                // Process with MAGI
                const lang = SettingsManager.getLanguage();
                const readingOrder = SettingsManager.getReadingOrder();

                console.log('Processing image:', img.src);
                const results = await magiProcessImage(img, lang, readingOrder);

                // Create overlays
                results.texts.forEach((text, idx) => {
                    const association = results.associations.find(a => a.textIndex === idx);
                    let speaker = null;

                    if (association) {
                        speaker = results.speakerMap[association.charIndex];
                    }

                    const rawText = results.ocrResults[idx] || '';
                    const langInfo = detectLanguageWithConfidence(rawText);
                    const cleanedText = cleanText(rawText, langInfo.lang);
                    const finalText = formatText(cleanedText);

                    // Only create overlay if text is meaningful
                    if (finalText.length > 0) {
                        createOverlay(
                            finalText,
                            text.bbox,
                            container,
                            speaker,
                            rawText,
                            img,
                            langInfo.lang
                        );
                    }
                });

            } catch (error) {
                console.error('Image processing failed:', error);
                // Remove processed flag to allow retry
                img.removeAttribute('data-magi-processed');

                // Don't retry immediately for server errors
                if (!error.message.includes('HTTP 5')) {
                    setTimeout(() => {
                        this.scheduleProcessing(img, 'high');
                    }, 5000);
                }
            }
        }

        cleanup() {
            // Clear all pending timeouts
            for (const timeout of this.pendingProcess.values()) {
                clearTimeout(timeout);
            }
            this.pendingProcess.clear();
            this.processingQueue.clear();
            this.currentlyProcessing.clear();
        }
    }

    function switchToEditMode(overlay, textContainer, editButton, img) {
        const textarea = document.createElement('textarea');
        textarea.className = 'ocr-edit';
        textarea.value = textContainer.textContent;
        textarea.style.cssText = `

            height: 80px;

            border-radius: 4px;

        textContainer.replaceWith(textarea);
        textarea.focus();

        // Create save button
        const saveButton = document.createElement('button');
        saveButton.className = 'save-btn';
        saveButton.textContent = 'Save';
        saveButton.style.cssText = `
            display: block;
            margin-top: 5px;
            padding: 6px 12px;
            background: #4CAF50;
            color: white;
            border: none;
            border-radius: 4px;
            cursor: pointer;
            font-size: 12px;
        `;
        overlay.appendChild(saveButton);

        // Create cancel button
        const cancelButton = document.createElement('button');
        cancelButton.className = 'cancel-btn';
        cancelButton.textContent = 'Cancel';
        cancelButton.style.cssText = `
            display: block;
            margin-top: 5px;
            padding: 6px 12px;
            background: #757575;
            color: white;
            border: none;
            border-radius: 4px;
            cursor: pointer;
            font-size: 12px;
            margin-left: 5px;
        `;
        overlay.appendChild(cancelButton);

        // Button container
        const buttonContainer = document.createElement('div');
        buttonContainer.style.display = 'flex';
        buttonContainer.style.justifyContent = 'flex-start';
        buttonContainer.style.gap = '5px';
        buttonContainer.appendChild(saveButton);
        buttonContainer.appendChild(cancelButton);
        overlay.appendChild(buttonContainer);

        // Remove edit button temporarily
        editButton.style.display = 'none';

        const saveHandler = () => {
            const correctedText = textarea.value.trim();
            const originalText = overlay.dataset.originalText;

            // Update display
            textContainer.textContent = correctedText;
            textarea.replaceWith(textContainer);
            buttonContainer.remove();
            editButton.style.display = 'block';

            // Send correction to backend if changed
            if (correctedText !== textContainer.textContent) {
                sendCorrection(img, overlay, correctedText, originalText);
            }

            // Update positioning after edit
            positioningSystem.updateOverlayPosition(overlay);
        };

        const cancelHandler = () => {
            textarea.replaceWith(textContainer);
            buttonContainer.remove();
            editButton.style.display = 'block';
            positioningSystem.updateOverlayPosition(overlay);
        };

        saveButton.addEventListener('click', saveHandler);
        cancelButton.addEventListener('click', cancelHandler);

        textarea.addEventListener('keydown', (e) => {
            if (e.key === 'Enter' && e.ctrlKey) saveHandler();
            if (e.key === 'Escape') cancelHandler();
        });
    }

    // Send correction to backend
    function sendCorrection(img, overlay, correctedText, originalText) {
        const bbox = JSON.parse(overlay.dataset.bbox);
        const lang = overlay.dataset.lang;

        // Get image data from cache or fetch
        let imageData = imageDataCache.get(img);

        const sendRequest = (base64) => {
            magiRequest(APP_CONFIG.MAGI_API.CORRECTION, {
                image: base64,
                bbox: bbox,
                original_text: originalText,
                corrected_text: correctedText,
                lang: lang
            }).then(response => {
                console.log('Correction saved:', response);
            }).catch(error => {
                console.error('Failed to save correction:', error);
            });
        };

        if (!imageData) {
            // Get image data and cache it
            getImageBase64(img).then(base64 => {
                imageDataCache.set(img, base64);
                sendRequest(base64);
            }).catch(error => {
                console.error('Failed to get image data for correction:', error);
            });
        } else {
            sendRequest(imageData);
        }
    }

            document.getElementById('order-manga').style.background = order === 'manga' ? '#4CAF50' : '#444';
            document.getElementById('order-western').style.background = order === 'western' ? '#4CAF50' : '#444';
        }

        async toggleOCR() {
            this.active = !this.active;
            SettingsManager.setActive(this.active);
            const btn = document.getElementById('magi-toggle');
            btn.textContent = this.active ? 'ON' : 'OFF';
            btn.style.background = this.active ? '#4CAF50' : '#f44336';

            if (this.active) {
                await this.enableOCR();
            } else {
                this.disableOCR();
            }
        }

        async enableOCR() {
            this.setupObservers();
            this.processExistingImages();
        }

        disableOCR() {
            this.removeAllOverlays();
            this.processedImages = new WeakSet();
            if (this.observer) this.observer.disconnect();
            if (this.lazyObserver) this.lazyObserver.disconnect();
            this.imageProcessor.cleanup();
        }

        setupObservers() {
            // Disconnect existing observers
            if (this.observer) this.observer.disconnect();
            if (this.lazyObserver) this.lazyObserver.disconnect();

            // Mutation observer for new images
            this.observer = new MutationObserver((mutations) => {
                for (const mutation of mutations) {
                    for (const node of mutation.addedNodes) {
                        if (node.nodeType === Node.ELEMENT_NODE) {
                            if (node.nodeName === 'IMG') {
                                this.imageProcessor.scheduleProcessing(node);
                            }
                            const images = node.querySelectorAll ? node.querySelectorAll('img') : [];
                            for (const img of images) {
                                this.imageProcessor.scheduleProcessing(img);
                            }
                        }
                    }
                }
            });

            this.observer.observe(document.body, {
                childList: true,
                subtree: true
            });

            // Lazy loading observer
            this.lazyObserver = new IntersectionObserver((entries) => {
                for (const entry of entries) {
                    if (entry.isIntersecting) {
                        this.imageProcessor.scheduleProcessing(entry.target, 'high');
                    }
                }
            }, {
                rootMargin: APP_CONFIG.PROCESSING.LAZY_LOAD_MARGIN,
                threshold: 0.1
            });

            // Observe all existing images
            document.querySelectorAll('img').forEach(img => {
                this.lazyObserver.observe(img);
            });
        }

        setupCleanup() {
            // Periodic cache cleanup
            this.cleanupInterval = setInterval(() => {
                imageDataCache.cleanup();
            }, APP_CONFIG.CACHE.CLEANUP_INTERVAL);

            // Cleanup on page unload
            window.addEventListener('beforeunload', () => {
                this.destroy();
            });
        }

        async processAllImages() {
            const images = document.querySelectorAll('img:not([data-magi-processed])');
            console.log(`Processing ${images.length} images...`);

            for (const img of images) {
                await this.imageProcessor.scheduleProcessing(img, 'high');
                // Small delay to prevent overwhelming the backend
                await new Promise(resolve => setTimeout(resolve, 100));
            }
        }

        processExistingImages() {
            if (!this.active) return;

            const images = document.querySelectorAll('img');
            images.forEach(img => {
                if (!img.hasAttribute('data-magi-processed') && img.complete && img.naturalHeight > 0) {
                    this.imageProcessor.scheduleProcessing(img);
                }
            });
        }

        removeAllOverlays() {
            document.querySelectorAll('.ocr-overlay').forEach(el => el.remove());
            document.querySelectorAll('[data-magi-processed]').forEach(img => {
                img.removeAttribute('data-magi-processed');
            });
            this.processedImages = new WeakSet();
        }

        destroy() {
            this.disableOCR();
            positioningSystem.destroy();
            connectionMonitor.stopPeriodicChecks();

            if (this.cleanupInterval) {
                clearInterval(this.cleanupInterval);
            }

            // Remove control panel
            const panel = document.getElementById('magi-control-panel');
            if (panel) panel.remove();
        }
    }
