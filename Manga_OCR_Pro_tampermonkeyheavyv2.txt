    // Enhanced Bulk Processing Manager with Auto-Processing
    class BulkProcessingManager {
    constructor() {
        this.isProcessing = false;
        this.progressCallbacks = [];
        this.batchQueue = [];
        this.autoProcessEnabled = true; // Auto-process by default
    }

    async autoProcessPageImages() {
        if (!this.autoProcessEnabled || this.isProcessing) {
            return;
        }

        try {
            this.isProcessing = true;
            console.log('Auto-processing page images...');

            const images = Array.from(document.querySelectorAll('img'))
                .filter(img => {
                    const rect = img.getBoundingClientRect();
                    return rect.width > 100 && rect.height > 100 &&
                           img.complete && img.naturalHeight !== 0 &&
                           !img.hasAttribute('data-magi-processed') &&
                           !img.hasAttribute('data-magi-processing');
                });

            if (images.length === 0) {
                console.log('No suitable images found for auto-processing');
                return;
            }

            console.log(`Auto-processing ${images.length} images`);

             // Process images individually to include auto-inpainting
             for (const img of images) {
                 try {
                     // ✅ PERMANENT LOCK: MARK AS PROCESSING IMMEDIATELY
                     // This prevents 5x duplicate requests while processing is running
                     img.setAttribute('data-magi-processing', 'true');

                     // Process the image (this will trigger auto-inpainting)
                     // v2.2: hand the image to the shared queue instead of a private processor
                     img.removeAttribute('data-magi-processing');
                     window.magiAppInstance.imageProcessor.scheduleProcessing(img);
                } catch (error) {
                    console.error(`Failed to process image ${img.src}:`, error);
                }
            }

            console.log('Auto-bulk processing completed');

        } catch (error) {
            console.error('Auto-bulk processing failed:', error);
        } finally {
            this.isProcessing = false;
        }
    }

        async processImageBatch(imageUrls, options = {}) {
            if (this.isProcessing) {
                throw new Error('Bulk processing already in progress');
            }

            this.isProcessing = true;

            try {
                const response = await magiRequestWithRetry(
                    APP_CONFIG.MAGI_API.PROCESS_BATCH,
                    {
                        image_paths: imageUrls,
                        output_dir: options.output_dir || 'bulk_output',
                        options: {
                            create_zip: options.create_zip !== false,
                            progress_callback: options.progress_callback,
                            reading_order: options.reading_order || 'manga',
                            auto_inpaint: true // Enable auto-inpainting
                        }
                    }
                );

                return response;
            } finally {
                this.isProcessing = false;
            }
        }

        async processPageImages(options = {}) {
            const images = Array.from(document.querySelectorAll('img'))
                .filter(img => {
                    const rect = img.getBoundingClientRect();
                    return rect.width > 100 && rect.height > 100 &&
                           img.complete && img.naturalHeight !== 0;
                })
                .map(img => img.src);

            if (images.length === 0) {
                throw new Error('No suitable images found on page');
            }

            return await this.processImageBatch(images, options);
        }

        addProgressCallback(callback) {
            this.progressCallbacks.push(callback);
        }

        removeProgressCallback(callback) {
            this.progressCallbacks = this.progressCallbacks.filter(cb => cb !== callback);
        }

        notifyProgress(progress, processed, total) {
            this.progressCallbacks.forEach(callback => {
                try {
                    callback(progress, processed, total);
                } catch (error) {
                    console.error('Progress callback error:', error);
                }
            });
        }
    }

    // Separate inpainting layer management
    async initializeInpaintingLayer() {

        `;

        return this.inpaintingLayer;
    }

    // Convert bounding boxes to proper format for backend
    normalizeBoundingBoxes(bboxes) {
        if (!Array.isArray(bboxes)) return [];

        return bboxes.map(bbox => {
            if (!Array.isArray(bbox)) return [];

            // Ensure all values are numbers
            return bbox.map(val => {
                const num = parseFloat(val);
                return isNaN(num) ? 0 : num;
            });
        }).filter(bbox => bbox.length === 4); // Only keep valid 4-element bboxes
    }

    async autoInpaintText(img, textBboxes, options = {}) {
        if (!this.autoInpaintingEnabled) {
            return;
        }

        try {
            console.log('Auto-inpainting text regions...');

            // Convert image to base64 first
            const imageBase64 = await getImageBase64(img);

            // Normalize bounding boxes to ensure they're numbers
            const normalizedBboxes = this.normalizeBoundingBoxes(textBboxes);

            if (normalizedBboxes.length === 0) {
                console.log('No valid bounding boxes found for inpainting');
                return;
            }

            // Prepare the data in the format the backend expects
            const requestData = {
                image: imageBase64,
                text_bboxes: normalizedBboxes,
                character_bboxes: this.normalizeBoundingBoxes(options.character_bboxes || []),
                translated_texts: options.translated_texts || [],
                method: 'auto',
                add_translations: false // Set to true if you want to add translations during inpainting
            };

            console.log('Sending inpainting request with data:', {
                image_length: imageBase64.length,
                text_bboxes_count: normalizedBboxes.length,
                text_bboxes_sample: normalizedBboxes.slice(0, 1) // Log first bbox for debugging
            });

            const response = await magiRequestWithRetry(
                APP_CONFIG.MAGI_API.INPAINT_TEXT,
                requestData
            );

            console.log('Auto-inpainting completed:', response);
            return response;
        } catch (error) {
            console.error('Auto-inpainting failed:', error);
            throw error;
        }
    }

    // Enhanced version that works with the actual image processing flow
    async autoInpaintImage(img, preDetectedBubbles = null) {
        if (!this.autoInpaintingEnabled) {
            return;
        }

        try {
            console.log('Starting auto-inpainting for image:', img.src);

            // First detect text regions (v2.2: reuse the detection from the main pass when we have it)
            const textBboxes = preDetectedBubbles || (await magiBackend.detectComponents(img)).bubbles || [];

            if (textBboxes.length === 0) {
                console.log('No text regions found for inpainting');
                return;
            }

            console.log(`Found ${textBboxes.length} text regions for inpainting`);

            // Log raw bbox data for debugging
            console.log('Raw bbox data sample:', textBboxes.slice(0, 1));

            // Perform inpainting
            const result = await this.autoInpaintText(img, textBboxes);

            if (result && result.result_image) {
                // Replace the image with the inpainted version
                img.src = result.result_image;

                // Store original source for reverting
                if (!img.dataset.originalSrc) {
                    img.dataset.originalSrc = img.src;
                }

                console.log('Image successfully inpainted');
            }

            return result;
        } catch (error) {
            console.error('Auto-inpainting image failed:', error);
        }
    }

        // NEW: Practical method to overlay text on an image element
        async overlayTextOnImage(imgElement, textData) {
            if (!this.useBackendOverlays) {
                console.log('Backend overlays disabled, using frontend method');
                return false;
            }

            try {
                // Convert image to base64
                const imageBase64 = await getImageBase64(imgElement);

                // Prepare text elements for backend
                const textElements = textData.map((item, index) => ({
                    original_text: item.originalText || '',
                    translated_text: item.translatedText || item.text,
                    bbox: item.bbox,
                    style: item.style || this.getDefaultStyle()
                }));

                // Use backend overlay
                const result = await this.createTextOverlay(imageBase64, textElements);

                if (result.overlay_image) {
                    // Replace the original image with the overlayed version
                    imgElement.src = result.overlay_image;

                    // Store original source for reverting
                    if (!imgElement.dataset.originalSrc) {
                        imgElement.dataset.originalSrc = imgElement.src;
                    }

                    return true;
                }
                return false;

            } catch (error) {
                console.error('Image overlay failed:', error);
                return false;
            }
        }

        // NEW: Revert to original image
        revertImageOverlay(imgElement) {
            if (imgElement.dataset.originalSrc) {
                imgElement.src = imgElement.dataset.originalSrc;
                delete imgElement.dataset.originalSrc;
                return true;
            }
            return false;
        }

        // NEW: Apply overlay to all processed images on page
        async applyOverlayToAllImages() {
            const images = document.querySelectorAll('img[data-magi-processed]');
            let successCount = 0;

            for (const img of images) {
                const container = img.closest('.magi-container');
                if (!container) continue;

                const bubbles = container.querySelectorAll('.manga-bubble');
                const textData = Array.from(bubbles).map(bubble => ({
                    text: bubble.querySelector('.bubble-text')?.textContent || '',
                    bbox: JSON.parse(bubble.dataset.bbox || '[]'),
                    originalText: bubble.dataset.originalText || '',
                    style: this.getBubbleStyle(bubble)
                }));

                if (textData.length > 0) {
                    const success = await this.overlayTextOnImage(img, textData);
                    if (success) {
                        successCount++;
                        // Hide frontend bubbles since text is baked in
                        bubbles.forEach(bubble => bubble.style.display = 'none');
                    }
                }
            }

            return successCount;
        }

                'short': 18,

    // Enhanced Translation Manager
    class TranslationManager {
        constructor() {
            this.translationCache = new Map();
            this.isTranslating = false;
            this.autoTranslate = GM_getValue('auto_translate', false);
            this.targetLanguage = GM_getValue('target_language', APP_CONFIG.DEFAULT_TRANSLATION_LANG);
            this.translationEnabled = GM_getValue('translation_enabled', true);
            this.translationStats = {
                totalTranslations: 0,
                cacheHits: 0,
                failedTranslations: 0,
                lastTranslationTime: null
            };
        }

        async translateText(text, sourceLanguage = 'auto') {
            if (!this.translationEnabled || !text || text.trim().length === 0) {
                return text;
            }

            const cacheKey = `${sourceLanguage}-${this.targetLanguage}-${text}`;
            if (this.translationCache.has(cacheKey)) {
                this.translationStats.cacheHits++;
                return this.translationCache.get(cacheKey);
            }

            try {
                this.isTranslating = true;
                const response = await magiRequestWithRetry(
                    APP_CONFIG.MAGI_API.TRANSLATE,
                    {
                        text: text,
                        source_language: sourceLanguage
                    }
                );

                if (response.error) {
                    console.warn('Translation failed:', response.error);
                    this.translationStats.failedTranslations++;
                    return text;
                }

                const translatedText = response.translated_text || text;
                this.translationCache.set(cacheKey, translatedText);
                this.translationStats.totalTranslations++;
                this.translationStats.lastTranslationTime = new Date();

                return translatedText;

            } catch (error) {
                console.error('Translation error:', error);
                this.translationStats.failedTranslations++;
                return text;
            } finally {
                this.isTranslating = false;
            }
        }

        async batchTranslateTexts(texts, sourceLanguage = 'auto') {
            if (!this.translationEnabled || !texts || texts.length === 0) {
                return texts;
            }

            // Filter out empty texts
            const validTexts = texts.filter(text => text && text.trim().length > 0);
            if (validTexts.length === 0) {
                return texts;
            }

            try {
                this.isTranslating = true;
                const response = await magiRequestWithRetry(
                    APP_CONFIG.MAGI_API.TRANSLATE_BATCH,
                    {
                        texts: validTexts,
                        source_language: sourceLanguage
                    }
                );

                if (response.error) {
                    console.warn('Batch translation failed:', response.error);
                    this.translationStats.failedTranslations += validTexts.length;
                    return texts;
                }

                const translatedTexts = response.translated_texts || texts;

                // Update cache with batch results
                validTexts.forEach((text, index) => {
                    const cacheKey = `${sourceLanguage}-${this.targetLanguage}-${text}`;
                    this.translationCache.set(cacheKey, translatedTexts[index]);
                });

                this.translationStats.totalTranslations += validTexts.length;
                this.translationStats.lastTranslationTime = new Date();

                return translatedTexts;

            } catch (error) {
                console.error('Batch translation error:', error);
                this.translationStats.failedTranslations += validTexts.length;
                return texts;
            } finally {
                this.isTranslating = false;
            }
        }

        async translateBubble(bubble) {

                        bubble.appendChild(indicator);

                    bubble.dataset.originalText = originalText;
                }
            }
        }

        // Language detection helper
        detectTextLanguage(text) {
            if (!text) return 'auto';

            const patterns = {
                'ja': [/[ぁ-ん]/, /[ァ-ン]/, /[一-龯]/], // Japanese
                'ko': /[\uAC00-\uD7AF]/, // Korean
                'zh': /[\u4E00-\u9FFF]/, // Chinese
                'ru': /[А-Яа-яЁё]/, // Russian
                'en': /[a-zA-Z]/ // English
            };

            let bestLang = 'auto';
            let bestScore = 0;

            for (const [lang, langPatterns] of Object.entries(patterns)) {
                let score = 0;
                const patternsArray = Array.isArray(langPatterns) ? langPatterns : [langPatterns];

                for (const pattern of patternsArray) {
                    const matches = text.match(pattern);
                    if (matches) score += matches.length;
                }

                if (score > bestScore) {
                    bestScore = score;
                    bestLang = lang;
                }
            }

            return bestScore > text.length * 0.3 ? bestLang : 'auto';
        }
    }

    // Enhanced ImageDataCache
    class ImageDataCache {
        constructor(maxSize = APP_CONFIG.CACHE.MAX_IMAGES) {
            this.cache = new WeakMap();
            this.keys = new Set();
            this.maxSize = maxSize;
            this.creationTimes = new WeakMap();
        }

        set(key, value) {
            if (this.keys.size >= this.maxSize) {
                const firstKey = this.keys.values().next().value;
                this.cache.delete(firstKey);
                this.creationTimes.delete(firstKey);
                this.keys.delete(firstKey);
            }
            this.cache.set(key, value);
            this.creationTimes.set(key, Date.now());
            this.keys.add(key);
        }

        get(key) {
            return this.cache.get(key);
        }

        cleanup(ttl = APP_CONFIG.CACHE.EMBEDDING_TTL) {
            const now = Date.now();
            for (const key of this.keys) {
                const creationTime = this.creationTimes.get(key);
                if (creationTime && (now - creationTime) > ttl) {
                    this.cache.delete(key);
                    this.creationTimes.delete(key);
                    this.keys.delete(key);
                }
            }
        }
    }

            return {
                healthy: this.isBackendHealthy,
                failures: this.failures,

                    this.responseTimes.reduce((a, b) => a + b, 0) / this.responseTimes.length : 0
            };
        }

    // Enhanced MAGI API request with retry logic
    async function magiRequestWithRetry(url, data, maxRetries = APP_CONFIG.PROCESSING.MAX_RETRIES) {
        for (let attempt = 1; attempt <= maxRetries; attempt++) {
            try {
                const result = await magiRequest(url, data);
                return result;
            } catch (error) {
                if (attempt === maxRetries) {
                    throw new Error(`MAGI request failed after ${maxRetries} attempts: ${error.message}`);
                }
                console.warn(`MAGI request failed, retrying... (${attempt}/${maxRetries})`);
                const backoffTime = 1000 * Math.pow(2, attempt - 1);
                await new Promise(resolve => setTimeout(resolve, backoffTime));
            }
        }
    }

    // Enhanced Overlay Positioning System with Stable Positioning
    class OverlayPositioningSystem {
        constructor() {
            this.overlays = new WeakMap();
            this.observedContainers = new WeakSet();
            this.resizeObserver = null;
            this.intersectionObserver = null;
            this.mutationObserver = null;
            this.scrollHandler = null;
            this.updateTimeouts = new WeakMap();
            this.globalUpdateTimeout = null;
            this.positionHistory = new WeakMap();
            this.collisionChecked = new WeakSet();
            this.stablePositions = new WeakMap();
        }

        init() {
            this.setupObservers();
            this.setupScrollHandler();
        }

        setupObservers() {
            this.resizeObserver = new ResizeObserver(entries => {
                entries.forEach(entry => {
                    const container = entry.target;
                    if (this.hasSignificantSizeChange(entry)) {
                        this.updateOverlaysInContainer(container);
                    }
                });
            });

            this.intersectionObserver = new IntersectionObserver(entries => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        const container = entry.target;
                        setTimeout(() => {
                            this.updateOverlaysInContainer(container);
                        }, 100);
                    }
                });
            }, {
                threshold: 0.1,
                rootMargin: '100px'
            });

            this.mutationObserver = new MutationObserver(mutations => {
                let needsUpdate = false;
                mutations.forEach(mutation => {
                    if (mutation.type === 'attributes') {
                        const target = mutation.target;
                        if (this.overlays.has(target)) {
                            this.debouncedUpdate(target);
                        }
                    } else if (mutation.addedNodes.length > 0 || mutation.removedNodes.length > 0) {
                        needsUpdate = true;
                    }
                });
                if (needsUpdate) {
                    this.checkForLayoutChanges();
                }
            });
        }

        hasSignificantSizeChange(entry) {
            const { width, height } = entry.contentRect;
            if (!entry.target.previousSize) {
                entry.target.previousSize = { width, height };
                return true;
            }

            const widthChange = Math.abs(width - entry.target.previousSize.width) / entry.target.previousSize.width;
            const heightChange = Math.abs(height - entry.target.previousSize.height) / entry.target.previousSize.height;

            entry.target.previousSize = { width, height };

            return widthChange > 0.05 || heightChange > 0.05;
        }

        setupScrollHandler() {
            let scrollTimeout;
            this.scrollHandler = () => {
                clearTimeout(scrollTimeout);
                scrollTimeout = setTimeout(() => {
                    this.debouncedGlobalUpdate();
                }, 250);
            };
            window.addEventListener('scroll', this.scrollHandler, { passive: true });
        }

        monitorOverlay(overlay, container, img, bbox, lang) {
            this.overlays.set(overlay, {
                container,
                img,
                bbox,
                lang,
                originalLeft: null,
                originalTop: null
            });

            if (!this.observedContainers.has(container)) {
                this.resizeObserver.observe(container);
                this.intersectionObserver.observe(container);
                this.mutationObserver.observe(container, {
                    attributes: true,
                    childList: true,
                    subtree: true,
                    attributeFilter: ['style', 'class', 'id']
                });
                this.observedContainers.add(container);
            }

            this.updateOverlayPosition(overlay);
        }

        updateOverlaysInContainer(container) {
            const overlays = container.querySelectorAll('.manga-bubble');
            overlays.forEach(overlay => {
                const data = this.overlays.get(overlay);
                if (data && data.container === container) {
                    this.updateOverlayPosition(overlay);
                }
            });
        }

        updateOverlayPosition(overlay) {
            const data = this.overlays.get(overlay);
            if (!data) return;

            const { container, img, bbox } = data;

            if (!document.contains(container) || !document.contains(img)) {
                this.overlays.delete(overlay);
                overlay.remove();
                return;
            }

            try {
                const imgRect = img.getBoundingClientRect();
                const containerRect = container.getBoundingClientRect();

                if (imgRect.width === 0 || imgRect.height === 0 ||
                    containerRect.width === 0 || containerRect.height === 0) {
                    return;
                }

                const scaleX = imgRect.width / Math.max(img.naturalWidth, 1);
                const scaleY = imgRect.height / Math.max(img.naturalHeight, 1);

                const scaledLeft = bbox[0] * imgRect.width;
                const scaledTop = bbox[1] * imgRect.height;
                const scaledWidth = Math.max((bbox[2] - bbox[0]) * imgRect.width, 60);
                const scaledHeight = Math.max((bbox[3] - bbox[1]) * imgRect.height, 40);

                const containerX = containerRect.left - imgRect.left;
                const containerY = containerRect.top - imgRect.top;

                const finalLeft = Math.max(0, scaledLeft + containerX);
                const finalTop = Math.max(0, scaledTop + containerY);

                const bubbleWidth = overlay.offsetWidth || scaledWidth;
                const finalWidth = Math.min(bubbleWidth, APP_CONFIG.UI.OVERLAY_MAX_WIDTH);

                // Store original position if not set
                if (data.originalLeft === null) {
                    data.originalLeft = finalLeft;
                    data.originalTop = finalTop;
                }

                // Check for stable position first
                const stablePosition = this.stablePositions.get(overlay);
                if (stablePosition && this.isPositionStable(overlay, stablePosition)) {
                    overlay.style.left = `${stablePosition.left}px`;
                    overlay.style.top = `${stablePosition.top}px`;
                } else {
                    // Use calculated position
                    overlay.style.left = `${finalLeft}px`;
                    overlay.style.top = `${finalTop}px`;
                }

                overlay.style.width = `${finalWidth}px`;
                overlay.style.minHeight = `${scaledHeight}px`;
                overlay.style.display = 'block';

                // Only run collision detection if needed
                if (!this.collisionChecked.has(overlay) || this.hasLayoutChanged(overlay)) {
                    const hadCollision = this.avoidElementCollisions(overlay, container);
                    if (hadCollision) {
                        // Store the stable position after collision resolution
                        this.stablePositions.set(overlay, {
                            left: parseFloat(overlay.style.left),
                            top: parseFloat(overlay.style.top)
                        });
                    }
                    this.collisionChecked.add(overlay);
                }

            } catch (error) {
                console.warn('Error updating overlay position:', error);
            }
        }

        isPositionStable(overlay, storedPosition) {
            const currentLeft = parseFloat(overlay.style.left) || 0;
            const currentTop = parseFloat(overlay.style.top) || 0;

            const positionDiff = Math.abs(currentLeft - storedPosition.left) +
                               Math.abs(currentTop - storedPosition.top);

            return positionDiff < 5;
        }

        hasLayoutChanged(overlay) {
            const data = this.overlays.get(overlay);
            if (!data) return true;

            const { container, img } = data;
            const currentContainerRect = container.getBoundingClientRect();
            const currentImgRect = img.getBoundingClientRect();

            if (!data.previousContainerRect) {
                data.previousContainerRect = currentContainerRect;
                data.previousImgRect = currentImgRect;
                return false;
            }

            const containerChanged =
                Math.abs(currentContainerRect.width - data.previousContainerRect.width) > data.previousContainerRect.width * 0.1 ||
                Math.abs(currentContainerRect.height - data.previousContainerRect.height) > data.previousContainerRect.height * 0.1;

            const imgChanged =
                Math.abs(currentImgRect.width - data.previousImgRect.width) > data.previousImgRect.width * 0.1 ||
                Math.abs(currentImgRect.height - data.previousImgRect.height) > data.previousImgRect.height * 0.1;

            data.previousContainerRect = currentContainerRect;
            data.previousImgRect = currentImgRect;

            return containerChanged || imgChanged;
        }

        avoidElementCollisions(overlay, container) {
            try {
                let hasCollision = false;
                let iterations = 0;
                const maxIterations = 14;

                // Force Directed Separation Algorithm
                while (iterations < maxIterations) {
                    const overlayRect = overlay.getBoundingClientRect();
                    let collisionFound = false;

                    // Spatial grid check for other bubbles
                    const otherBubbles = Array.from(document.querySelectorAll('.manga-bubble'))
                        .filter(bubble => bubble !== overlay && this.isBubbleVisible(bubble));

                    for (const otherBubble of otherBubbles) {
                        const otherRect = otherBubble.getBoundingClientRect();

                        // ADD EXTRA 4px PADDING BETWEEN BUBBLES
                        const expandedOtherRect = {
                            left: otherRect.left - 4,
                            top: otherRect.top - 4,
                            right: otherRect.right + 4,
                            bottom: otherRect.bottom + 4
                        };

                        if (this.elementsOverlap(overlayRect, expandedOtherRect)) {
                            // Calculate repulsion vector
                            const repulsion = this.calculateRepulsionForce(overlayRect, otherRect);
                            this.applyForceToBubble(overlay, repulsion, container);
                            collisionFound = true;
                            hasCollision = true;
                        }
                    }

                    // ONLY RUN PAGE ELEMENT CHECK ON FIRST ITERATION
                    if (iterations === 0) {
                        // Check against page elements
                        const centerX = overlayRect.left + overlayRect.width / 2;
                        const centerY = overlayRect.top + overlayRect.height / 2;
                        const pageElements = document.elementsFromPoint(centerX, centerY);

                        for (const element of pageElements) {
                            if (element === overlay || element === container ||
                                element.contains(overlay) || overlay.contains(element) ||
                                element.classList.contains('manga-bubble') ||
                                element.classList.contains('magi-container')) {
                                continue;
                            }

                            const elementRect = element.getBoundingClientRect();
                            if (this.elementsOverlap(overlayRect, elementRect)) {
                                // Move bubble away from page element
                                const resolution = this.findSafePositionNearby(overlay, elementRect, container);
                                if (resolution.resolved) {
                                    collisionFound = true;
                                    hasCollision = true;
                                }
                            }
                        }
                    }

                    if (!collisionFound) break;
                    iterations++;
                }

                // Apply soft damping to prevent oscillation
                this.dampenBubblePosition(overlay);

                if (hasCollision) {
                    overlay.style.opacity = '0.95';
                    // NEVER USE STABLE POSITION CACHE FOR COLLIDED BUBBLES
                    this.stablePositions.delete(overlay);
                } else {
                    this.stablePositions.delete(overlay);
                    overlay.style.opacity = '';
                }

                return hasCollision;

            } catch (error) {
                console.warn('Error in collision avoidance:', error);
                return false;
            }
        }

        calculateRepulsionForce(rect1, rect2) {
            const centerX1 = rect1.left + rect1.width / 2;
            const centerY1 = rect1.top + rect1.height / 2;
            const centerX2 = rect2.left + rect2.width / 2;
            const centerY2 = rect2.top + rect2.height / 2;

            let dx = centerX1 - centerX2;
            let dy = centerY1 - centerY2;

            // Minimum force to ensure separation
            const distance = Math.sqrt(dx*dx + dy*dy) || 1;
            const force = Math.max(60, 420 / distance);

            dx = dx / distance * force;
            dy = dy / distance * force;

            return { dx, dy };
        }

        applyForceToBubble(overlay, force, container) {
            const containerRect = container.getBoundingClientRect();
            const currentLeft = parseFloat(overlay.style.left) || 0;
            const currentTop = parseFloat(overlay.style.top) || 0;

            let newLeft = currentLeft + force.dx;
            let newTop = currentTop + force.dy;

            // Keep inside container bounds
            const overlayWidth = overlay.offsetWidth;
            const overlayHeight = overlay.offsetHeight;

            newLeft = Math.max(0, Math.min(newLeft, containerRect.width - overlayWidth - 10));
            newTop = Math.max(0, Math.min(newTop, containerRect.height - overlayHeight - 10));

            overlay.style.left = `${newLeft}px`;
            overlay.style.top = `${newTop}px`;
        }

        findSafePositionNearby(overlay, obstacleRect, container) {
            const containerRect = container.getBoundingClientRect();
            const overlayRect = overlay.getBoundingClientRect();

            const positions = [
                { x: overlayRect.left, y: obstacleRect.top - overlayRect.height - 20 },
                { x: overlayRect.left, y: obstacleRect.bottom + 20 },
                { x: obstacleRect.right + 20, y: overlayRect.top },
                { x: obstacleRect.left - overlayRect.width - 20, y: overlayRect.top },
                { x: overlayRect.left - 30, y: overlayRect.top },
                { x: overlayRect.left + 30, y: overlayRect.top },
                { x: overlayRect.left, y: overlayRect.top - 30 },
                { x: overlayRect.left, y: overlayRect.top + 30 },
            ];

            for (const pos of positions) {
                const testRect = {
                    left: pos.x,
                    top: pos.y,
                    right: pos.x + overlayRect.width,
                    bottom: pos.y + overlayRect.height
                };

                if (testRect.left >= containerRect.left &&
                    testRect.top >= containerRect.top &&
                    testRect.right <= containerRect.right &&
                    testRect.bottom <= containerRect.bottom) {

                    // Convert to container relative coordinates
                    const relativeX = pos.x - containerRect.left;
                    const relativeY = pos.y - containerRect.top;

                    overlay.style.left = `${Math.max(0, relativeX)}px`;
                    overlay.style.top = `${Math.max(0, relativeY)}px`;

                    return { resolved: true };
                }
            }

            return { resolved: false };
        }

        dampenBubblePosition(overlay) {
            const data = this.overlays.get(overlay);
            if (!data) return;

            const currentLeft = parseFloat(overlay.style.left) || 0;
            const currentTop = parseFloat(overlay.style.top) || 0;

            // Soft pull back towards original position
            if (data.originalLeft !== null) {
                const dampening = 0.05;
                const newLeft = currentLeft + (data.originalLeft - currentLeft) * dampening;
                const newTop = currentTop + (data.originalTop - currentTop) * dampening;

                overlay.style.left = `${newLeft}px`;
                overlay.style.top = `${newTop}px`;
            }
        }

        isBubbleVisible(bubble) {
            const rect = bubble.getBoundingClientRect();
            return (
                rect.top >= 0 &&
                rect.left >= 0 &&
                rect.bottom <= (window.innerHeight || document.documentElement.clientHeight) &&
                rect.right <= (window.innerWidth || document.documentElement.clientWidth)
            );
        }

        adjustBubblePosition(overlay, obstacleRect, container) {
            const overlayRect = overlay.getBoundingClientRect();
            const containerRect = container.getBoundingClientRect();
            const data = this.overlays.get(overlay);

            const strategies = [
                { x: data.originalLeft, y: data.originalTop, priority: 0 },
                { x: overlayRect.left, y: obstacleRect.top - overlayRect.height - 25, priority: 1 },
                { x: overlayRect.left, y: obstacleRect.bottom + 25, priority: 2 },
                { x: obstacleRect.right + 25, y: overlayRect.top, priority: 3 },
                { x: obstacleRect.left - overlayRect.width - 25, y: overlayRect.top, priority: 4 },
                { x: containerRect.left + 15, y: containerRect.top + 15, priority: 5 },
                { x: containerRect.right - overlayRect.width - 15, y: containerRect.top + 15, priority: 6 }
            ];

            strategies.sort((a, b) => a.priority - b.priority);

            for (const position of strategies) {
                if (this.isPositionValid(position, overlayRect, containerRect)) {
                    const relativeX = position.x;
                    const relativeY = position.y;

                    overlay.style.left = `${relativeX}px`;
                    overlay.style.top = `${relativeY}px`;

                    return { resolved: true, strategy: position.priority };
                }
            }

            return { resolved: false };
        }

        isPositionValid(position, overlayRect, containerRect) {
            const testRect = {
                left: position.x,
                top: position.y,
                right: position.x + overlayRect.width,
                bottom: position.y + overlayRect.height
            };

            const withinContainer = (
                testRect.left >= containerRect.left &&
                testRect.top >= containerRect.top &&
                testRect.right <= containerRect.right &&
                testRect.bottom <= containerRect.bottom
            );

            const partiallyInViewport = (
                testRect.right > 0 &&
                testRect.left < window.innerWidth &&
                testRect.bottom > 0 &&
                testRect.top < window.innerHeight
            );

            return withinContainer && partiallyInViewport;
        }

        elementsOverlap(rect1, rect2) {

                rect1.right < rect2.left ||
                rect1.left > rect2.right ||
                rect1.bottom < rect2.top ||
                rect1.top > rect2.bottom
            );
        }

                method: 'POST',
                url: url,
                data: JSON.stringify(data),

                ontimeout: () => reject(new Error('Request timeout'))
            });
        });
    }

    // Enhanced image validation and base64 conversion
    // ---- v2.2: encode each image once and share it between detect / embed / OCR ----
    const base64Cache = new WeakMap();
    function getImageBase64(img) {
        if (!img || !img.src) {
            return Promise.reject(new Error('Invalid image element'));
        }
        const cached = base64Cache.get(img);
        if (cached && cached.src === img.src) return cached.promise;
        const promise = getImageBase64Uncached(img);
        base64Cache.set(img, { src: img.src, promise });
        promise.catch(() => {
            const c = base64Cache.get(img);
            if (c && c.promise === promise) base64Cache.delete(img);
        });
        return promise;
    }
    function releaseImageBase64(img) {
        base64Cache.delete(img);
    }

    async function getImageBase64Uncached(img) {
        if (!img || !img.src) {
            throw new Error('Invalid image element');
        }

        if (img.src.startsWith('data:') ||
            img.src.startsWith('blob:') ||
            new URL(img.src, window.location.href).origin === location.origin) {

            return new Promise((resolve, reject) => {
                if (!img.complete || img.naturalHeight === 0) {
                    reject(new Error('Image not loaded'));
                    return;
                }

                try {
                    const canvas = document.createElement('canvas');
                    const ctx = canvas.getContext('2d');
                    canvas.width = img.naturalWidth;
                    canvas.height = img.naturalHeight;
                    ctx.drawImage(img, 0, 0);
                    resolve(canvas.toDataURL('image/png'));
                } catch (error) {
                    reject(new Error(`Canvas conversion failed: ${error.message}`));
                }
            });
        }

        return new Promise((resolve, reject) => {
            GM_xmlhttpRequest({
                method: 'POST',
                url: APP_CONFIG.CORS_PROXY,
                data: JSON.stringify({ imageUrl: img.src }),
                headers: {'Content-Type': 'application/json'},
                timeout: 15000,
                onload: (res) => {
                    try {
                        const data = JSON.parse(res.responseText);
                        if (data.base64) {
                            resolve(data.base64);
                        } else {
                            reject(new Error('No base64 data in response'));
                        }
                    } catch (e) {
                        reject(new Error(`Proxy response parsing failed: ${e.message}`));
                    }
                },
                onerror: (err) => {
                    reject(new Error(`CORS proxy failed: ${err.statusText}`));
                },
                ontimeout: () => {
                    reject(new Error('CORS proxy timeout'));
                }
            });
        });
    }

    // Enhanced Character Mapper with persistence
    class CharacterMapper {
        constructor() {
            this.clusterToSpeaker = {};
            this.nextSpeakerId = 1;
            this.storedEmbeds = {};
            this.loadFromStorage();
        }

        loadFromStorage() {
            try {
                const stored = GM_getValue('character_mapper_data');
                if (stored) {
                    const data = JSON.parse(stored);
                    this.clusterToSpeaker = data.clusterToSpeaker || {};
                    this.nextSpeakerId = data.nextSpeakerId || 1;
                    this.storedEmbeds = data.storedEmbeds || {};
                }
            } catch (error) {
                console.warn('Failed to load character mapper from storage:', error);
            }
        }

        saveToStorage() {
            try {
                GM_setValue('character_mapper_data', JSON.stringify({
                    clusterToSpeaker: this.clusterToSpeaker,
                    nextSpeakerId: this.nextSpeakerId,
                    storedEmbeds: this.storedEmbeds
                }));
            } catch (error) {
                console.warn('Failed to save character mapper to storage:', error);
            }
        }

        getSpeaker(embeddings) {
            const globalId = this.findBestMatch(embeddings);

            if (!this.clusterToSpeaker[globalId]) {
                this.clusterToSpeaker[globalId] = `CHAR${this.nextSpeakerId++}`;
                this.saveToStorage();
            }
            return this.clusterToSpeaker[globalId];
        }

        findBestMatch(embedding) {
            if (!Array.isArray(embedding) || embedding.length === 0) {
                return this.createNewCharacter(embedding);
            }

            if (Object.keys(this.storedEmbeds).length === 0) {
                const newId = 1;
                this.storedEmbeds[newId] = embedding;
                this.saveToStorage();
                return newId;
            }

            let bestId = null;
            let bestSimilarity = 0;

            for (const [id, storedEmbeds] of Object.entries(this.storedEmbeds)) {
                const similarity = this.cosineSimilarity(embedding, storedEmbeds);
                if (similarity > bestSimilarity && similarity > APP_CONFIG.SIMILARITY_THRESHOLD) {
                    bestSimilarity = similarity;
                    bestId = parseInt(id);
                }
            }

            if (bestId !== null) {
                return bestId;
            }

            return this.createNewCharacter(embedding);
        }

        createNewCharacter(embedding) {
            const newId = Object.keys(this.storedEmbeds).length + 1;
            this.storedEmbeds[newId] = embedding;
            this.saveToStorage();
            return newId;
        }

        cosineSimilarity(embedding, storedEmbeds) {
            if (!Array.isArray(embedding) || !Array.isArray(storedEmbeds)) {
                return 0;
            }
            if (embedding.length !== storedEmbeds.length) {
                return 0;
            }

            let dot = 0;
            for (let i = 0; i < embedding.length; i++) {
                dot += embedding[i] * storedEmbeds[i];
            }

            let magA = 0;
            let magB = 0;
            for (let i = 0; i < embedding.length; i++) {
                magA += embedding[i] * embedding[i];
                magB += storedEmbeds[i] * storedEmbeds[i];
            }
            magA = Math.sqrt(magA);
            magB = Math.sqrt(magB);

            if (magA === 0 || magB === 0) {
                return 0;
            }

            return dot / (magA * magB);
        }

        clear() {
            this.clusterToSpeaker = {};
            this.nextSpeakerId = 1;
            this.storedEmbeds = {};
            GM_deleteValue('character_mapper_data');
        }
    }

    // Enhanced language detection
    function detectLanguageWithConfidence(text) {
        if (!text || text.length < 3) return { lang: 'eng', confidence: 0 };

        const patterns = {
            'jpn': [/[ぁ-ん]/, /[ァ-ン]/, /[一-龯]/],
            'kor': [/[\uAC00-\uD7AF]/],
            'chi_sim': [/[\u4E00-\u9FFF]/],
            'ru': [/[А-Яа-яЁё]/],
            'eng': [/[a-zA-Z]/]
        };

        let bestLang = 'eng';
        let bestScore = 0;

        for (const [lang, langPatterns] of Object.entries(patterns)) {
            let score = 0;
            for (const pattern of langPatterns) {
                const matches = text.match(pattern);
                if (matches) score += matches.length;
            }

            if (score > bestScore) {
                bestScore = score;
                bestLang = lang;
            }
        }

        const confidence = Math.min(bestScore / text.length, 1);
        return { lang: bestLang, confidence };
    }

    // Text processing utilities - NO WORD BREAKING
    function cleanText(text, lang) {
        if (typeof text !== 'string') return '';

        const basePattern = 'A-Za-z0-9\\s.,!?\'"\\-—「」『』（）«»„""';
        let langRanges = '';

        switch (lang) {
            case 'ru':
                langRanges = '\\u0400-\\u04FF';
                break;
            case 'chi_sim':
                langRanges = '\\u4E00-\\u9FFF';
                break;
            case 'jpn':
                langRanges = [
                    '\\u3040-\\u309F',
                    '\\u30A0-\\u30FF',
                    '\\u31F0-\\u31FF',
                    '\\u4E00-\\u9FFF',
                    '\\uFF66-\\uFF9D'
                ].join('');
                break;
            case 'kor':
                langRanges = '\\uAC00-\\uD7AF';
                break;
            default:
                langRanges = '';
        }

        const whitelistRegex = new RegExp(`[^${basePattern}${langRanges}]`, 'gu');
        const multiSpaceRegex = /\s+/g;

        return text
            .replace(whitelistRegex, ' ')
            .replace(multiSpaceRegex, ' ')
            .trim();
    }

    function formatText(text) {
        if (!text) return '';
        const trimmed = text.trim();
        if (trimmed.length === 0) return '';
        return trimmed.charAt(0).toUpperCase() + trimmed.slice(1);
    }

    // === Better approach - format based on source language only ===
    function formatMangaText(text, originalLang, targetLang = 'en') {
        if (!text || text.length === 0) return text;

        const mangaFormatting = {
            // Japanese text (regardless of target language)
            'ja': { maxLineLength: 15 },
            'jpn': { maxLineLength: 15 },

            // Korean text
            'ko': { maxLineLength: 18 },
            'kor': { maxLineLength: 18 },

            // Chinese text
            'zh': { maxLineLength: 12 },
            'chi_sim': { maxLineLength: 12 },
            'chi_tra': { maxLineLength: 12 },

            // Default for other languages
            'default': { maxLineLength: 20 }
        };

        // Use source language for formatting rules
        const rules = mangaFormatting[originalLang] || mangaFormatting['default'];

        console.log(`Formatting ${originalLang} text with max ${rules.maxLineLength} chars per line`);

        let formattedText = text;
        formattedText = applyNaturalLineBreaks(formattedText, rules.maxLineLength);
        formattedText = applyMangaEmphasis(formattedText);
        formattedText = applyMangaSpacing(formattedText);

        return formattedText
            .split('\n')
            .map(line => line.trim())
            .filter(line => line.length > 0)
            .join('\n');
    }

    // === Natural line breaking (no word cutting) ===
    function applyNaturalLineBreaks(text, maxLineLength) {
        // Split text by punctuation that ends sentences, but keep punctuation
        const sentences = text.split(/([.!?]+)\s*/g);
        let result = [];

        for (let i = 0; i < sentences.length; i += 2) {
            const sentence = (sentences[i] || '') + (sentences[i + 1] || '');
            const words = sentence.trim().split(/\s+/);
            let currentLine = [];

            for (let word of words) {
                const linePreview = [...currentLine, word].join(' ');
                if (linePreview.length > maxLineLength) {
                    // Push the current line and start a new one
                    result.push(currentLine.join(' '));
                    currentLine = [word];
                } else {
                    currentLine.push(word);
                }
            }

            if (currentLine.length > 0) result.push(currentLine.join(' '));
        }

        // Join all lines with newlines for clean display
        return result.join('\n').trim();
    }

    // === Emphasis & emotional punctuation styling ===
    function applyMangaEmphasis(text) {
        // Add a bit of spacing around strong punctuation for manga-like look
        return text
            .replace(/([!?]{1,3})/g, ' $1 ')
            .replace(/\s+/g, ' ')
            .trim();
    }

    // === Spacing fix around punctuation and words ===
    function applyMangaSpacing(text) {
        return text
            // Ensure punctuation has space before next word
            .replace(/([,.!?])([A-Za-z])/g, '$1 $2')
            // Remove extra spaces but keep line breaks
            .split('\n')
            .map(line => line.replace(/\s+/g, ' ').trim())
            .join('\n');
    }

    // Calculate optimal bubble size based on text content
    function calculateTextMetrics(text) {
        const lines = text.split('\n');
        const maxLineLength = Math.max(...lines.map(line => line.length));

        const charWidth = 8;

        };
    }

    // Text-character association
    function associateTextToCharacters(texts, characters, panels, readingOrder = "manga") {
        const associations = [];

        panels.forEach(panel => {
            const panelBbox = panel.bbox;
            const panelChars = characters.filter(char => isInside(char.bbox, panelBbox));
            const panelTexts = texts.filter(text => isInside(text.bbox, panelBbox));

            panelTexts.forEach(text => {
                let minDist = Infinity;
                let closestChar = null;

                panelChars.forEach(char => {
                    let dist = bboxDistance(text.bbox, char.bbox);

                    if (readingOrder === "manga") {
                        const textCenter = (text.bbox[0] + text.bbox[2]) / 2;
                        const charCenter = (char.bbox[0] + char.bbox[2]) / 2;
                        if (textCenter > charCenter) dist *= 0.7;
                    }

                    if (dist < minDist) {
                        minDist = dist;
                        closestChar = char;
                    }
                });

                if (closestChar) {
                    associations.push({
                        textIndex: texts.indexOf(text),
                        charIndex: characters.indexOf(closestChar)
                    });
                }
            });
        });

        return associations;
    }

    function isInside(inner, outer) {
        return (
            inner[0] >= outer[0] &&
            inner[1] >= outer[1] &&
            inner[2] <= outer[2] &&
            inner[3] <= outer[3]
        );
    }

    function bboxDistance(a, b) {
        const aCenter = [(a[0] + a[2]) / 2, (a[1] + a[3]) / 2];
        const bCenter = [(b[0] + b[2]) / 2, (b[1] + b[3]) / 2];
        return Math.sqrt(
            Math.pow(aCenter[0] - bCenter[0], 2) +
            Math.pow(aCenter[1] - bCenter[1], 2)
        );
    }

        // With /get_all_data both results are already here. Otherwise OCR and embeddings are sent
        // together and overlays only wait for OCR (speaker badges are added when embeddings return).
        const showSpeakers = APP_CONFIG.PERF.SHOW_SPEAKER_BADGES;
        const embeddingsPromise = !showSpeakers ? Promise.resolve({ char_embeddings: [] })
            : (bundledEmb ? Promise.resolve(bundledEmb)
                          // text_bboxes are required: the server returns nothing without them
                          : magiBackend.getCharacterEmbeddings(img, charBboxes, textBboxes));
        const ocrPromise = bundledOcr ? Promise.resolve({ results: bundledOcr })
                                      : magiBackend.ocrTextRegions(img, textBboxes, lang);

        const speakerMap = {};
        const speakerReady = embeddingsPromise.then(embeddings => {
            const charEmbeddings = (embeddings && embeddings.char_embeddings) || [];
            const mapper = getCharacterMapper();
            for (let i = 0; i < normalizedCharacters.length; i++) {
                if (charEmbeddings[i]) {
                    speakerMap[i] = mapper.getSpeaker(charEmbeddings[i]);
                }
            }
            return speakerMap;
        }).catch(err => {
            speakerStatus.failed = true;
            console.warn('Speaker embeddings failed (overlays are not affected):', err);
            return speakerMap;
        });

        const associations = associateTextToCharacters(normalizedTexts, normalizedCharacters, panels, readingOrder);

        const ocrResults = await ocrPromise;
        const ocrArr = ocrResults.results || [];

        // TOP OF PAGE FIRST SORTING.
        // v2.4 fix: the backend returns boxes in confidence order, and OCR text + speaker associations
        // are indexed in that order. The old code sorted only the boxes, so after sorting, box N could be
        // shown with the text that belongs to a different box. Sort a permutation and apply it to all three.
        const sortCompare = (a, b) => {
            // PRIMARY SORT: ALWAYS TOP OF PAGE FIRST (smallest Y coordinate = highest on page)
            const yDiff = a.bbox[1] - b.bbox[1];

            // Only when almost same vertical level (<3% page height), use manga right-to-left order
            if (Math.abs(yDiff) < 0.03) {
                return b.bbox[0] - a.bbox[0];
            }

            return yDiff;
        };
        const order = normalizedTexts.map((_, i) => i)
            .sort((i, j) => sortCompare(normalizedTexts[i], normalizedTexts[j]));
        const newIndexOf = new Array(order.length);
        order.forEach((oldIdx, newIdx) => { newIndexOf[oldIdx] = newIdx; });

        return {
            texts: order.map(i => normalizedTexts[i]),
            characters: normalizedCharacters,
            associations: associations.map(a => ({ ...a, textIndex: newIndexOf[a.textIndex] })),
            ocrResults: order.map(i => (ocrArr[i] === undefined || ocrArr[i] === null) ? '' : ocrArr[i]),
            speakerMap,
            speakerReady,
            speakerStatus,
            detectionBubbles: texts
        };
    } catch (error) {
        console.error('MAGI processing failed:', error);
        throw error;
    }
}

   function ensureMagiContainer(img) {
        let container = img.parentElement;
        if (!container.classList.contains('magi-container')) {
            container = document.createElement('div');
            container.className = 'magi-container';

   // Numbers the page images 1..N in document order and tracks which page you are on and which
   // way you are scrolling. The queue uses this: current page first, then pages ahead of you,
   // with pages behind you interleaved at a penalty (like a media player's read-ahead buffer).
   class PageTracker {
        constructor() {
            this.pages = [];
            this.numbers = new WeakMap();
            this.current = 1;
            this.direction = 1;          // 1 = reading down, -1 = scrolling back up
            this.lastProbeTop = null;
            this.dirty = true;
            this.refreshTimer = null;
            this.scrollRaf = null;
            this.onChange = null;

            const onScroll = () => {
                if (this.scrollRaf) return;
                this.scrollRaf = requestAnimationFrame(() => {
                    this.scrollRaf = null;
                    this.updateCurrent();
                });
            };
            // capture: also catches sites that scroll an inner container instead of the window
            document.addEventListener('scroll', onScroll, { capture: true, passive: true });
            window.addEventListener('resize', () => this.markDirty());
        }

        isPageCandidate(img) {
            if (!img.isConnected || img.closest('#magi-control-panel, #magi-hud')) return false;
            const r = img.getBoundingClientRect();
            if (r.width < 100) return false;
            if (img.complete && img.naturalHeight > 0) {
                const min = APP_CONFIG.PERF.MIN_IMAGE_SIZE;
                return img.naturalWidth >= min && img.naturalHeight >= min;
            }
            return r.height >= 150;      // still loading: keep its slot so later numbers do not shift
        }

        markDirty() {
            this.dirty = true;
            clearTimeout(this.refreshTimer);
            this.refreshTimer = setTimeout(() => this.ensureFresh(), 200);
        }

        ensureFresh() {
            if (!this.dirty) return;
            this.dirty = false;
            this.refresh();
        }

        refresh() {
            const pages = [];
            document.querySelectorAll('img').forEach(img => {
                if (this.isPageCandidate(img)) pages.push(img);
            });
            this.pages = pages;
            pages.forEach((img, i) => {
                this.numbers.set(img, i + 1);
                const badge = img.parentElement && img.parentElement.querySelector(':scope > .magi-page-badge');
                if (badge && badge.dataset.n !== String(i + 1)) badge.dataset.n = String(i + 1);
            });
            this.updateCurrent();
        }

        updateCurrent() {
            const pages = this.pages;
            if (!pages.length) { this.current = 1; return; }

            // direction from how the first page moved on screen since the last check
            const top0 = pages[0].getBoundingClientRect().top;
            if (this.lastProbeTop !== null) {
                const dy = top0 - this.lastProbeTop;
                if (Math.abs(dy) >= 10) this.direction = dy < 0 ? 1 : -1;
            }
            this.lastProbeTop = top0;

            const vh = window.innerHeight;
            let best = 0, bestVis = 0, nearest = 1, nearestDist = Infinity;
            for (let i = 0; i < pages.length; i++) {
                const r = pages[i].getBoundingClientRect();
                const vis = Math.min(r.bottom, vh) - Math.max(r.top, 0);
                if (vis > bestVis) { bestVis = vis; best = i + 1; }
                const dist = Math.abs((r.top + r.bottom) / 2 - vh / 2);
                if (dist < nearestDist) { nearestDist = dist; nearest = i + 1; }
            }
            this.current = best || nearest;

            // Pages that are not loaded yet cannot be processed: make the next few eager
            const ahead = APP_CONFIG.PERF.PRELOAD_AHEAD;
            const dir = this.direction >= 0 ? 1 : -1;
            for (let k = 1; k <= ahead; k++) {
                const p = pages[this.current - 1 + k * dir];
                if (p && p.loading === 'lazy') p.loading = 'eager';
            }

            if (this.onChange) this.onChange();
        }

        // lower = processed sooner
        score(img) {
            const n = this.numbers.get(img);
            if (!n) return 1e6;
            const ahead = this.direction >= 0 ? n - this.current : this.current - n;
            let s = ahead >= 0 ? ahead : -ahead * APP_CONFIG.PERF.BEHIND_PENALTY + 0.5;
            const r = img.getBoundingClientRect();
            if (r.bottom > 0 && r.top < window.innerHeight) s -= 1000;   // on screen right now
            return s;
        }

        setState(img, state) {
            if (!APP_CONFIG.PERF.SHOW_PAGE_BADGES) return;
            const container = ensureMagiContainer(img);
            let badge = container.querySelector(':scope > .magi-page-badge');
            if (!badge) {
                badge = document.createElement('div');
                badge.className = 'magi-page-badge notranslate';
                badge.setAttribute('translate', 'no');
                badge.title = 'Click to (re)process this page';
                container.appendChild(badge);
            }
            // data-* only: the positioning system's observers ignore these, so no re-layout is triggered
            badge.dataset.state = state;
            badge.dataset.n = String(this.numbers.get(img) || '');
        }
   }

   // Remembers detection+OCR per image URL (+language/reading order) across reloads.
   const ResultCache = {
        map: null,
        timer: null,
        load() {
            if (this.map) return;
            try {
                this.map = new Map(JSON.parse(GM_getValue(APP_CONFIG.PERF.RESULT_CACHE_KEY, '[]')));
            } catch (e) {
                this.map = new Map();
            }
        },
        key(img, lang, order) {
            const s = img.src;
            if (!s || s.startsWith('data:') || s.startsWith('blob:')) return null;
            return `${s}|${lang}|${order}`;
        },
        get(key) {
            if (!key) return null;
            this.load();
            const e = this.map.get(key);
            if (!e) return null;
            if (Date.now() - e.t > APP_CONFIG.PERF.RESULT_CACHE_TTL_MS) { this.map.delete(key); return null; }
            this.map.delete(key);        // refresh LRU position
            this.map.set(key, e);
            return e.v;
        },
        set(key, value) {
            if (!key) return;
            this.load();
            this.map.delete(key);
            this.map.set(key, { t: Date.now(), v: value });
            while (this.map.size > APP_CONFIG.PERF.RESULT_CACHE_MAX) {
                this.map.delete(this.map.keys().next().value);
            }
            clearTimeout(this.timer);
            this.timer = setTimeout(() => {
                try { GM_setValue(APP_CONFIG.PERF.RESULT_CACHE_KEY, JSON.stringify([...this.map])); }
                catch (e) { console.warn('Result cache not saved:', e); }
            }, 3000);
        }
   };

            let s = `${connectionMonitor.isBackendHealthy ? '🟢' : '🔴'} Page ${pt.current}/${pt.pages.length}  ✓ ${ip.stats.done}`;
            if (ip.running.size) s += `  ⚙ ${ip.running.size}`;
            if (ip.queue.size) s += `  ⏳ ${ip.queue.size}`;
            if (ip.stats.cached) s += `  ⚡ ${ip.stats.cached}`;
            if (ip.stats.failed) s += `  ✗ ${ip.stats.failed}`;
            const ft = ip.finishTimes;
            if (ft.length >= 3 && ip.running.size + ip.queue.size > 0) {
                const secPerPage = (ft[ft.length - 1] - ft[0]) / (ft.length - 1) / 1000;
                const left = (ip.running.size + ip.queue.size) * secPerPage;
                const fmt = (x) => x < 60 ? `${Math.round(x)}s` : `${Math.floor(x / 60)}m${String(Math.round(x % 60)).padStart(2, '0')}s`;
                s += `  ${secPerPage.toFixed(1)}s/pg  ETA ${fmt(left)}`;
            }
            if (ip.paused) s += '  (paused)';
            this.text.textContent = s;
            this.btn.textContent = ip.paused ? '▶' : '⏸';
            this.el.classList.toggle('idle', !busy && !ip.paused);
        }
   }

   const IMAGE_PRIORITY_RANK = {
        'now': -1,
        'highest': 0,
        'high': 1,
        'medium': 2,
        'near-above': 3,
        'near-below': 4,
        'far-above': 5,
        'far-below': 6
    };

   // v2.2: queue-driven processor. No timers/staggering: the best-placed image (visible first,
   // then nearest to the viewport) starts the moment a slot is free.
   class ImageProcessor {
    constructor(app) {
        this.app = app;
        this.queue = new Map();        // img -> forced priority (or null)
        this.running = new Map();      // img -> job promise
        this.inpaintQueue = [];        // background inpainting, after overlays are shown
        this.inpaintRunning = 0;
        this.retryCounts = new WeakMap();
        this.pumpTimer = null;
        this.maxConcurrent = APP_CONFIG.PROCESSING.MAX_CONCURRENT_PROCESSING;
        this.baseConcurrent = this.maxConcurrent;
        this.autoFeaturesEnabled = true;
        this.mousePosition = { x: 0, y: 0 };

        this.paused = false;
        this.stats = { done: 0, failed: 0, cached: 0 };
        this.bestMs = Infinity;
        this.goodStreak = 0;
        this.finishTimes = [];         // recent completion times, for s/page + ETA

        // legacy names kept so nothing external breaks
        this.pendingProcess = new Map();
        this.processingQueue = new Set();
        this.currentlyProcessing = this.running;

        this.setupMouseTracking();
    }

    setupMouseTracking() {
        document.addEventListener('mousemove', (e) => {
            this.mousePosition.x = e.clientX;
            this.mousePosition.y = e.clientY;
        }, { passive: true });
    }

    // Get processing priority for an image
    getImagePriority(img, rect = img.getBoundingClientRect()) {
        const isInViewport = !(
            rect.bottom < 0 ||
            rect.top > window.innerHeight ||
            rect.right < 0 ||
            rect.left > window.innerWidth
        );

        const isNearMouse = this.isPointNearRect(this.mousePosition, rect, 200);

        if (isInViewport && isNearMouse) {
            return 'highest';
        } else if (isInViewport) {
            return 'high';
        } else if (isNearMouse) {
            return 'medium';
        }
        return this.getDistanceFromViewport(rect);
    }

    isPointNearRect(point, rect, threshold) {
        const distanceX = Math.max(rect.left - point.x, point.x - rect.right, 0);
        const distanceY = Math.max(rect.top - point.y, point.y - rect.bottom, 0);
        return Math.sqrt(distanceX * distanceX + distanceY * distanceY) <= threshold;
    }

    getDistanceFromViewport(rect) {
        if (rect.bottom < 0) {
            return (0 - rect.bottom) < 1000 ? 'near-above' : 'far-above';
        } else if (rect.top > window.innerHeight) {
            return (rect.top - window.innerHeight) < 1000 ? 'near-below' : 'far-below';
        }
        return 'high';
    }

    // Can this image be processed right now?
    isEligible(img) {
        if (!img || img.nodeName !== 'IMG' || !img.isConnected) return false;
        if (img.hasAttribute('data-magi-processed')) return false;
        if (!img.src || !img.complete || img.naturalHeight === 0) return false;
        const min = APP_CONFIG.PERF.MIN_IMAGE_SIZE;
        if (img.naturalWidth < min || img.naturalHeight < min) return false;
        return true;
    }

    // Add an image to the queue (cheap; safe to call repeatedly)
    async scheduleProcessing(img, forcePriority = null) {
        if (!this.isEligible(img) || this.running.has(img)) return;

        if (!this.queue.has(img)) {
            this.queue.set(img, forcePriority);
            if (this.app.pageTracker) this.app.pageTracker.setState(img, 'queued');
        } else if (forcePriority) {
            const prev = this.queue.get(img);
            if (!prev || IMAGE_PRIORITY_RANK[forcePriority] < IMAGE_PRIORITY_RANK[prev]) {
                this.queue.set(img, forcePriority);
            }
        }
        if (this.app.pageTracker) this.app.pageTracker.markDirty();
        this.schedulePump();
    }

    schedulePump(delay = 0) {
        if (this.pumpTimer) return;
        // setTimeout(0) lets a burst of MutationObserver/IntersectionObserver calls coalesce,
        // so the whole burst is ordered by priority in one go.
        this.pumpTimer = setTimeout(() => {
            this.pumpTimer = null;
            this.pump();
        }, delay);
    }

    // Choose the best queued image *now* (you may have scrolled/jumped since it was queued):
    // on-screen pages first, current page first among them, then pages ahead of you in order,
    // with pages behind you interleaved at BEHIND_PENALTY.
    pickNext() {
        const pt = this.app.pageTracker;
        let best = null;
        let bestScore = Infinity;

        for (const [img, forced] of this.queue) {
            if (!this.isEligible(img)) {
                this.queue.delete(img); // not loaded yet / gone / done - a 'load' event re-queues it
                continue;
            }
            const score = forced === 'now' ? -1e9 : (pt ? pt.score(img) : 0);
            if (score < bestScore) {
                best = img;
                bestScore = score;
            }
        }
        return best;
    }

    togglePause() {
        this.paused = !this.paused;
        if (!this.paused) this.pump();
        if (this.app.hud) this.app.hud.render();
    }

    // Adaptive concurrency (AIMD-style): add a slot while jobs stay near their best observed time,
    // drop one when they slow right down, halve on failures.
    recordLatency(ms, ok) {
        if (!APP_CONFIG.PERF.ADAPTIVE_CONCURRENCY) return;
        if (!ok) {
            this.maxConcurrent = Math.max(1, Math.floor(this.maxConcurrent / 2));
            this.goodStreak = 0;
            return;
        }
        this.bestMs = Math.min(this.bestMs, ms);
        if (ms <= this.bestMs * 1.3) {
            if (++this.goodStreak >= 3 && this.queue.size > 0 &&
                this.maxConcurrent < APP_CONFIG.PERF.MAX_CONCURRENT_CAP) {
                this.maxConcurrent++;
                this.goodStreak = 0;
            }
        } else if (ms > this.bestMs * 2.5) {
            this.maxConcurrent = Math.max(this.baseConcurrent, this.maxConcurrent - 1);
            this.goodStreak = 0;
        } else {
            this.goodStreak = 0;
        }
    }

    pump() {
        if (!this.app.active) return;
        if (this.app.pageTracker) this.app.pageTracker.ensureFresh();   // page numbers must be current before choosing
        if (this.paused) {
            if (this.app.hud) this.app.hud.update();
            return;
        }

        if (this.queue.size && !connectionMonitor.isBackendHealthy) {
            this.schedulePump(2000);
            return;
        }

        while (this.running.size < this.maxConcurrent && this.queue.size) {
            const img = this.pickNext();
            if (!img) break;
            this.queue.delete(img);
            this.startJob(img);
        }

        this.pumpInpaint();
        if (this.app.hud) this.app.hud.update();
    }

    startJob(img) {
        const started = performance.now();
        if (this.app.pageTracker) this.app.pageTracker.setState(img, 'working');

        const job = (async () => {
            let outcome = { ok: false, cached: false };
            try {
                outcome = await this.processImageInternal(img);
            } finally {
                releaseImageBase64(img);
                this.running.delete(img);
                if (outcome && !outcome.cached) {
                    this.recordLatency(performance.now() - started, outcome.ok);
                    if (outcome.ok) {
                        this.finishTimes.push(performance.now());
                        if (this.finishTimes.length > 8) this.finishTimes.shift();
                    }
                }
                this.pump();
            }
        })();
        this.running.set(img, job);
    }

    // Legacy entry point
    async processImage(img) {
        return this.scheduleProcessing(img, 'high');
    }

    // Background inpainting: only when no OCR work is waiting for a slot
    pumpInpaint() {
        if (!this.autoFeaturesEnabled || this.paused) return;
        if (this.inpaintRunning >= APP_CONFIG.PERF.MAX_CONCURRENT_INPAINT) return;
        if (this.queue.size > 0 || this.running.size >= this.maxConcurrent) return;

        const task = this.inpaintQueue.shift();
        if (!task) return;

        this.inpaintRunning++;
        this.autoInpaintImage(task.img, task.bubbles)
            .catch(() => {})
            .finally(() => {
                this.inpaintRunning--;
                releaseImageBase64(task.img);
                this.pumpInpaint();
            });
    }

    processExistingImages() {
        if (!this.app.active) return;
        document.querySelectorAll('img:not([data-magi-processed])').forEach(img => {
            this.scheduleProcessing(img);
        });
    }

    async processAllImagesWithPriority() {
        if (!this.app.active) return;
        const images = document.querySelectorAll('img:not([data-magi-processed])');
        console.log(`Queueing ${images.length} images (visible first)...`);
        images.forEach(img => this.scheduleProcessing(img, 'high'));
    }

    sortImagesByPriority(images) {
        return images.sort((a, b) =>
            IMAGE_PRIORITY_RANK[this.getImagePriority(a)] - IMAGE_PRIORITY_RANK[this.getImagePriority(b)]);
    }

    async processAllImages() {
        await this.processAllImagesWithPriority();
    }

    async autoInpaintImage(img, preDetectedBubbles = null) {
        try {
            const overlayManager = getOverlayManager();
            await overlayManager.autoInpaintImage(img, preDetectedBubbles);
        } catch (error) {
            console.error('Auto-inpainting failed:', error);
        }
    }

    async processImageInternal(img) {
        try {
            const container = ensureMagiContainer(img);

            container.querySelectorAll('.manga-bubble').forEach(el => el.remove());

            img.setAttribute('data-magi-processed', 'true');

            const lang = SettingsManager.getLanguage();
            const readingOrder = SettingsManager.getReadingOrder();

            console.log('Processing image:', img.src);
            // v2.3: instant when this exact image was already processed (reload / revisit)
            const cacheKey = APP_CONFIG.PERF.RESULT_CACHE ? ResultCache.key(img, lang, readingOrder) : null;
            let results = cacheKey ? ResultCache.get(cacheKey) : null;
            const fromCache = !!results;
            if (fromCache) {
                results = { ...results, speakerReady: Promise.resolve(results.speakerMap || {}) };
                this.stats.cached++;
            } else {
                results = await magiProcessImage(img, lang, readingOrder);
            }

            // Google Translate state does not change per bubble: look it up once per image
            const gtLang = detectGoogleTranslateLanguage();

            const created = [];
            results.texts.forEach((text, idx) => {
                const association = results.associations.find(a => a.textIndex === idx);
                let speaker = null;

                if (association) {
                    speaker = results.speakerMap[association.charIndex] || null;
                }

                const rawText = results.ocrResults[idx] || '';
                const effectiveLang = gtLang || detectLanguageWithConfidence(rawText).lang;
                const cleanedText = cleanText(rawText, effectiveLang);
                const finalText = formatText(cleanedText);

                if (finalText.length > 0) {
                    const bubble = createMangaBubble(
                        finalText,
                        text.bbox,
                        container,
                        speaker,
                        rawText,
                        img,
                        effectiveLang
                    );
                    if (bubble) created.push({ bubble, idx });
                }
            });

            this.stats.done++;
            if (this.app.pageTracker) this.app.pageTracker.setState(img, created.length ? 'done' : 'empty');

            // Save for next time once speakers are known (skip incomplete results so they get retried)
            if (cacheKey && !fromCache && results.speakerReady) {
                results.speakerReady.then(sm => {
                    if (results.speakerStatus && results.speakerStatus.failed) return;
                    if (!results.ocrResults.length && results.texts.length) return;
                    ResultCache.set(cacheKey, {
                        texts: results.texts,
                        associations: results.associations,
                        ocrResults: results.ocrResults,
                        speakerMap: sm,
                        detectionBubbles: results.detectionBubbles
                    });
                });
            }

            // Speaker badges are filled in when the embeddings call returns (overlays never wait for it)
            if (results.speakerReady && created.length) {
                results.speakerReady.then(speakerMap => {
                    for (const { bubble, idx } of created) {
                        if (!bubble.isConnected) continue;
                        const a = results.associations.find(x => x.textIndex === idx);
                        const sp = a ? speakerMap[a.charIndex] : null;
                        if (sp) setBubbleSpeaker(bubble, sp);
                    }
                }).catch(() => {});
            }

            // Apply backend overlay if enabled
            const overlayManager = getOverlayManager();
            if (overlayManager.useBackendOverlays) {
                setTimeout(async () => {
                    await applyBackendOverlayToContainer(container);
                }, 1500);
            }

            // Inpainting now happens in the background, after the overlays are already on screen
            if (this.autoFeaturesEnabled && overlayManager.autoInpaintingEnabled &&
                Array.isArray(results.detectionBubbles) && results.detectionBubbles.length) {
                this.inpaintQueue.push({ img, bubbles: results.detectionBubbles });
            }

            return { ok: true, cached: fromCache };

        } catch (error) {
            console.error('Image processing failed:', error);
            img.removeAttribute('data-magi-processed');
            this.stats.failed++;
            if (this.app.pageTracker) this.app.pageTracker.setState(img, 'failed');

            const attempts = (this.retryCounts.get(img) || 0) + 1;
            this.retryCounts.set(img, attempts);

            if (!String(error.message).includes('HTTP 5') && attempts <= APP_CONFIG.PERF.MAX_IMAGE_RETRIES) {
                setTimeout(() => {
                    this.scheduleProcessing(img, 'high');
                }, 2000 * attempts);
            }
            return { ok: false, cached: false };
        }
    }

    cleanup() {
        clearTimeout(this.pumpTimer);
        this.pumpTimer = null;
        this.queue.clear();
        this.inpaintQueue.length = 0;
        this.pendingProcess.clear();
        this.processingQueue.clear();
    }
}

    function getEffectiveLanguage(rawText) {
        // First, check if Google Translate is active
        const googleTranslateLang = detectGoogleTranslateLanguage();
        if (googleTranslateLang) {
            console.log('Using Google Translate language:', googleTranslateLang);
            return googleTranslateLang;
        }

        // Fall back to OCR-based language detection
        const langInfo = detectLanguageWithConfidence(rawText);
        console.log('Using OCR-detected language:', langInfo.lang);
        return langInfo.lang;
    }

    // Monitor for Google Translate changes
    function setupGoogleTranslateMonitor() {
        // Observe DOM changes for Google Translate elements
        const observer = new MutationObserver((mutations) => {
            let shouldUpdate = false;

            mutations.forEach((mutation) => {
                mutation.addedNodes.forEach((node) => {
                    if (node.nodeType === Node.ELEMENT_NODE) {
                        if (node.hasAttribute('translated') ||
                            node.querySelector('[translated]') ||
                            node.src?.includes('translate.google')) {
                            shouldUpdate = true;
                        }
                    }
                });
            });

            if (shouldUpdate) {
                console.log('Google Translate detected, updating bubbles...');
                // Re-process images to use the new language
                setTimeout(() => {
                    if (window.magiAppInstance) {
                        window.magiAppInstance.processExistingImages();
                    }
                }, 1000);
            }
        });

        observer.observe(document.body, {
            childList: true,
            subtree: true,
            attributes: true,
            attributeFilter: ['translated', 'lang']
        });

        return observer;
    }

    function normalizeAPIResponse(response) {
    if (!response) return response;

    // Handle bounding boxes in detection responses
    if (response.bubbles && Array.isArray(response.bubbles)) {
        response.bubbles = response.bubbles.map(bubble => {
            if (bubble.bbox && Array.isArray(bubble.bbox)) {
                return {
                    ...bubble,
                    bbox: bubble.bbox.map(val => {
                        const num = parseFloat(val);
                        return isNaN(num) ? 0 : num;
                    })
                };
            }
            return bubble;
        });
    }

    if (response.characters && Array.isArray(response.characters)) {
        response.characters = response.characters.map(character => {
            if (character.bbox && Array.isArray(character.bbox)) {
                return {
                    ...character,
                    bbox: character.bbox.map(val => {
                        const num = parseFloat(val);
                        return isNaN(num) ? 0 : num;
                    })
                };
            }
            return character;
        });
    }

    if (response.panels && Array.isArray(response.panels)) {
        response.panels = response.panels.map(panel => {
            if (panel.bbox && Array.isArray(panel.bbox)) {
                return {
                    ...panel,
                    bbox: panel.bbox.map(val => {
                        const num = parseFloat(val);
                        return isNaN(num) ? 0 : num;
                    })
                };
            }
            return panel;
        });
    }

    return response;
}


    // ---- v2.2 helpers ----
    // /register_bubble used to fire one request per bubble while detect/OCR were still queued on the
    // same backend. Now it is sent once the processing queue is idle.
    const bubbleRegistrationQueue = [];
    let bubbleRegistrationTimer = null;

    function queueBubbleRegistration(data) {
        bubbleRegistrationQueue.push(data);
        scheduleBubbleRegistrationFlush();
    }

    function scheduleBubbleRegistrationFlush() {
        clearTimeout(bubbleRegistrationTimer);
        bubbleRegistrationTimer = setTimeout(flushBubbleRegistrations, APP_CONFIG.PERF.REGISTER_BUBBLES_DELAY_MS);
    }

    function flushBubbleRegistrations() {
        const ip = window.magiAppInstance && window.magiAppInstance.imageProcessor;
        if (ip && (ip.queue.size > 0 || ip.running.size > 0)) {
            scheduleBubbleRegistrationFlush();
            return;
        }
        bubbleRegistrationQueue.splice(0).forEach(data => {
            fetch('http://localhost:5000/register_bubble', {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify(data)
            }).catch(console.error);
        });
    }

    function setBubbleSpeaker(bubble, speaker) {
        const content = bubble.querySelector('.bubble-content');
        if (!content || content.querySelector('.speaker-badge')) return;
        const speakerEl = document.createElement('div');
        speakerEl.className = 'speaker-badge';
        speakerEl.textContent = speaker;
        speakerEl.setAttribute('translate', 'no');
        content.insertBefore(speakerEl, content.firstChild);
        if (bubble._regData) bubble._regData.speaker = speaker;
        positioningSystem.debouncedUpdate(bubble);
    }

    function createMangaBubble(text, bbox, container, speaker, rawText, img, lang) {
    const imgElement = container.querySelector('img');
    if (!imgElement) return null;

    // Normalize the bbox for this bubble
    const normalizedBbox = Array.isArray(bbox) ? bbox.map(val => {
        const num = parseFloat(val);
        return isNaN(num) ? 0 : num;
    }) : [0, 0, 0, 0];

    const formattedText = formatMangaText(text, lang, 'en');
    const textMetrics = calculateTextMetrics(formattedText);
    const lengthCategory = getTextLengthCategory(formattedText);

    const bubbleData = {
        original_text: rawText,
        bbox: normalizedBbox,
        container_info: container.className || 'unknown',
        image_src: img.src,
        source_language: lang,
        speaker: speaker
    };

    // Send to backend for tracking (deferred until the backend is idle)
    queueBubbleRegistration(bubbleData);

    const bubble = document.createElement('div');
    bubble.className = 'manga-bubble';
    bubble._regData = bubbleData;
    bubble.dataset.originalText = rawText;
    bubble.dataset.bbox = JSON.stringify(normalizedBbox); // Store normalized bbox
    bubble.dataset.lang = lang;
    bubble.dataset.textLength = lengthCategory;
    bubble.dataset.hidden = 'false';

        // Set language attributes for translation
        bubble.setAttribute('translate', 'yes');
        bubble.setAttribute('lang', lang);

        bubble.style.minWidth = textMetrics.minWidth + 'px';
        bubble.style.maxWidth = textMetrics.maxWidth + 'px';

        // Create buttons
        const hideButton = document.createElement('button');
        hideButton.className = 'bubble-hide-btn';
        hideButton.innerHTML = '<i class="fas fa-eye"></i>';
        hideButton.title = 'Hide/Show bubble';
        hideButton.setAttribute('translate', 'no');
        bubble.appendChild(hideButton);

        const editButton = document.createElement('button');
        editButton.className = 'bubble-edit-btn';
        editButton.innerHTML = '<i class="fas fa-edit"></i>';
        editButton.title = 'Edit translation';
        editButton.setAttribute('translate', 'no');
        bubble.appendChild(editButton);

        // Add translate button
        const translateButton = document.createElement('button');
        translateButton.className = 'bubble-translate-btn';
        translateButton.innerHTML = '<i class="fas fa-language"></i>';
        translateButton.title = 'Translate bubble';
        translateButton.setAttribute('translate', 'no');
        bubble.appendChild(translateButton);

        // ADD BACKEND OVERLAY BUTTON
        const backendOverlayButton = document.createElement('button');
        backendOverlayButton.className = 'bubble-backend-overlay-btn';
        backendOverlayButton.innerHTML = '<i class="fas fa-image"></i>';
        backendOverlayButton.title = 'Bake text into image (backend overlay)';
        backendOverlayButton.setAttribute('translate', 'no');
        bubble.appendChild(backendOverlayButton);

        // Hidden indicator
        const hiddenIndicator = document.createElement('div');
        hiddenIndicator.className = 'bubble-hidden-indicator';
        bubble.appendChild(hiddenIndicator);

        // Content
        const bubbleContent = document.createElement('div');
        bubbleContent.className = 'bubble-content';

        if (speaker) {
            const speakerEl = document.createElement('div');
            speakerEl.className = 'speaker-badge';
            speakerEl.textContent = speaker;
            speakerEl.setAttribute('translate', 'no');
            bubbleContent.appendChild(speakerEl);
        }

        const textContainer = document.createElement('div');
        textContainer.className = 'bubble-text';
        textContainer.textContent = formattedText;
        textContainer.setAttribute('translate', 'yes');
        textContainer.setAttribute('lang', lang);
        bubbleContent.appendChild(textContainer);

        bubble.appendChild(bubbleContent);
        container.appendChild(bubble);

        positioningSystem.monitorOverlay(bubble, container, imgElement, bbox, lang);

        // Event listeners
        hideButton.addEventListener('click', (e) => {
            e.stopPropagation();
            toggleBubbleVisibility(bubble, hideButton);
        });

        editButton.addEventListener('click', (e) => {
            e.stopPropagation();
            switchToEditMode(bubble, textContainer, editButton, img);
        });

        translateButton.addEventListener('click', async (e) => {
            e.stopPropagation();
            await translateSingleBubble(bubble, textContainer, translateButton);
        });

        // NEW: Backend overlay event listener
        backendOverlayButton.addEventListener('click', async (e) => {
            e.stopPropagation();
            await applyBackendOverlayToBubble(bubble, imgElement, textContainer);
        });

        // Auto-translate if enabled
        const tm = getTranslationManager();
        if (tm.autoTranslate && tm.translationEnabled) {
            setTimeout(() => {
                tm.translateBubble(bubble);
            }, 500);
        }

        // NEW: Auto-apply backend overlay if enabled
        const overlayManager = getOverlayManager();
        if (overlayManager.useBackendOverlays) {
            setTimeout(async () => {
                await applyBackendOverlayToBubble(bubble, imgElement, textContainer);
            }, 1000);
        }

        return bubble;
    }

    // NEW: Function to apply backend overlay to a single bubble
    async function applyBackendOverlayToBubble(bubble, imgElement, textContainer) {
        const overlayManager = getOverlayManager();

        try {
            const textData = [{
                text: textContainer.textContent,
                translatedText: textContainer.textContent,
                originalText: bubble.dataset.originalText,
                bbox: JSON.parse(bubble.dataset.bbox),
                style: overlayManager.getBubbleStyle(bubble)
            }];

            const success = await overlayManager.overlayTextOnImage(imgElement, textData);

            if (success) {
                // Hide this bubble since text is now baked into image
                bubble.style.display = 'none';

                // Show success indicator
                bubble.classList.add('backend-overlay-applied');

                console.log('Backend overlay applied successfully to bubble');
            } else {
                console.warn('Failed to apply backend overlay to bubble');
            }

        } catch (error) {
            console.error('Error applying backend overlay to bubble:', error);
        }
    }

    // NEW: Function to apply backend overlay to all bubbles in a container
    async function applyBackendOverlayToContainer(container) {
        const imgElement = container.querySelector('img');
        if (!imgElement) return false;

        const bubbles = container.querySelectorAll('.manga-bubble:not(.backend-overlay-applied)');
        if (bubbles.length === 0) return false;

        const overlayManager = getOverlayManager();
        const textData = [];

        // Collect all bubble data
        bubbles.forEach(bubble => {
            const textContainer = bubble.querySelector('.bubble-text');
            if (textContainer) {
                textData.push({
                    text: textContainer.textContent,
                    translatedText: textContainer.textContent,
                    originalText: bubble.dataset.originalText,
                    bbox: JSON.parse(bubble.dataset.bbox || '[]'),
                    style: overlayManager.getBubbleStyle(bubble)
                });
            }
        });

        if (textData.length === 0) return false;

        try {
            const success = await overlayManager.overlayTextOnImage(imgElement, textData);

            if (success) {
                // Hide all bubbles in this container
                bubbles.forEach(bubble => {
                    bubble.style.display = 'none';
                    bubble.classList.add('backend-overlay-applied');
                });

                console.log(`Applied backend overlay to ${bubbles.length} bubbles in container`);
                return true;
            }

            return false;

        } catch (error) {
            console.error('Error applying backend overlay to container:', error);
            return false;
        }
    }

    // Translate single bubble
    async function translateSingleBubble(bubble, textContainer, translateButton) {
        const tm = getTranslationManager();
        const originalText = textContainer.textContent;
        const sourceLang = bubble.dataset.lang || 'auto';

        translateButton.disabled = true;
        translateButton.innerHTML = '<i class="fas fa-spinner fa-spin"></i>';

        try {
            const translatedText = await tm.translateText(originalText, sourceLang);
            if (translatedText !== originalText) {
                textContainer.textContent = translatedText;
                bubble.classList.add('translated');

                // Add translation indicator if not present
                if (!bubble.querySelector('.translation-indicator')) {
                    const indicator = document.createElement('div');
                    indicator.className = 'translation-indicator';
                    indicator.title = `Translated from ${sourceLang} to ${tm.targetLanguage}`;
                    bubble.appendChild(indicator);
                }
            }
        } catch (error) {
            console.error('Failed to translate bubble:', error);
        } finally {
            translateButton.disabled = false;
            translateButton.innerHTML = '<i class="fas fa-language"></i>';
        }
    }

    function detectGoogleTranslateLanguage() {
        // Check for Google Translate elements and attributes
        const translatedElements = document.querySelectorAll('[translated]');
        if (translatedElements.length > 0) {
            // Get the language from the first translated element
            const firstTranslated = translatedElements[0];
            const lang = firstTranslated.getAttribute('lang') ||
                        firstTranslated.closest('[lang]')?.getAttribute('lang');

            if (lang) {
                console.log('Google Translate detected, language:', lang);
                return lang;
            }
        }

        // Check for Google Translate iframe
        const googleTranslateFrame = document.querySelector('iframe[src*="translate.google"]');
        if (googleTranslateFrame) {
            console.log('Google Translate iframe detected');
            return navigator.language || 'en'; // Use browser language as fallback
        }

        // Check for Google Translate widget
        const googleTranslateWidget = document.querySelector('#google_translate_element');
        if (googleTranslateWidget) {
            console.log('Google Translate widget detected');
            return navigator.language || 'en';
        }

        return null; // No Google Translate detected
    }

    // Toggle bubble visibility
    function toggleBubbleVisibility(bubble, hideButton) {
        console.log('Toggling bubble visibility');

        const isHidden = bubble.dataset.hidden === 'true';

        if (isHidden) {
            // Show bubble
            bubble.dataset.hidden = 'false';
            if (hideButton) {
                hideButton.innerHTML = '<i class="fas fa-eye"></i>';
                hideButton.title = 'Hide bubble';
            }
            console.log('Bubble shown');
        } else {
            // Hide bubble
            bubble.dataset.hidden = 'true';
            if (hideButton) {
                hideButton.innerHTML = '<i class="fas fa-eye-slash"></i>';
                hideButton.title = 'Show bubble';
            }
            console.log('Bubble hidden');
        }

        // Force reflow to ensure CSS updates
        bubble.style.display = 'none';
        bubble.offsetHeight; // Trigger reflow
        bubble.style.display = 'block';
    }

    function switchToEditMode(bubble, textContainer, editButton, img) {
        const textarea = document.createElement('textarea');
        textarea.className = 'bubble-edit';
        textarea.value = textContainer.textContent;
        textarea.setAttribute('translate', 'yes');
        textarea.setAttribute('lang', bubble.dataset.lang);
        textContainer.replaceWith(textarea);
        textarea.focus();

        const saveButton = document.createElement('button');
        saveButton.className = 'bubble-save-btn';
        saveButton.textContent = 'Save';
        saveButton.setAttribute('translate', 'no');
        bubble.appendChild(saveButton);

        const cancelButton = document.createElement('button');
        cancelButton.className = 'bubble-cancel-btn';
        cancelButton.textContent = 'Cancel';
        cancelButton.setAttribute('translate', 'no');
        bubble.appendChild(cancelButton);

        const buttonContainer = document.createElement('div');
        buttonContainer.className = 'bubble-button-container';
        buttonContainer.appendChild(saveButton);
        buttonContainer.appendChild(cancelButton);
        bubble.appendChild(buttonContainer);

        editButton.style.display = 'none';

        const saveHandler = () => {
            const correctedText = textarea.value.trim();
            const originalText = bubble.dataset.originalText;

            textContainer.textContent = correctedText;
            textarea.replaceWith(textContainer);
            buttonContainer.remove();
            saveButton.remove();
            cancelButton.remove();
            editButton.style.display = 'block';

            if (correctedText !== originalText) {
                sendCorrection(img, bubble, correctedText, originalText);
            }

            positioningSystem.updateOverlayPosition(bubble);
        };

        const cancelHandler = () => {
            textarea.replaceWith(textContainer);
            buttonContainer.remove();
            saveButton.remove();
            cancelButton.remove();
            editButton.style.display = 'block';
            positioningSystem.updateOverlayPosition(bubble);
        };

        saveButton.addEventListener('click', saveHandler);
        cancelButton.addEventListener('click', cancelHandler);

        textarea.addEventListener('keydown', (e) => {
            if (e.key === 'Enter' && e.ctrlKey) saveHandler();
            if (e.key === 'Escape') cancelHandler();
        });
    }

    // Send correction to backend
    function sendCorrection(img, bubble, correctedText, originalText) {
        const bbox = JSON.parse(bubble.dataset.bbox);
        const lang = bubble.dataset.lang;

        let imageData = imageDataCache.get(img);

        const sendRequest = (base64) => {
            magiRequest(APP_CONFIG.MAGI_API.CORRECTION, {
                image: base64,
                bbox: bbox,
                original_text: originalText,
                corrected_text: correctedText,
                lang: lang
            }).then(response => {
                console.log('Correction saved:', response);
            }).catch(error => {
                console.error('Failed to save correction:', error);
            });
        };

        if (!imageData) {
            getImageBase64(img).then(base64 => {
                imageDataCache.set(img, base64);
                sendRequest(base64);
            }).catch(error => {
                console.error('Failed to get image data for correction:', error);
            });
        } else {
            sendRequest(imageData);
        }
    }

        // Add drag functionality for minimized panel
        setupDragFunctionality() {
            const panel = document.getElementById('magi-control-panel');
            let isDragging = false;
            let dragOffset = { x: 0, y: 0 };

            // Function to check if we should ignore drag for this element
            const shouldIgnoreDrag = (element) => {
                // Don't drag when clicking on these interactive elements
                const ignoreElements = [
                    'button', 'input', 'select', 'textarea', 'a'
                ];

                return ignoreElements.some(tagName =>
                    element.tagName?.toLowerCase() === tagName ||
                    element.closest?.(tagName)
                );
            };

            panel.addEventListener('mousedown', (e) => {
                // Don't start drag if clicking on interactive elements
                if (shouldIgnoreDrag(e.target)) {
                    return;
                }

                // Allow drag in BOTH minimized and unminimized states
                isDragging = true;
                dragOffset.x = e.clientX - panel.getBoundingClientRect().left;
                dragOffset.y = e.clientY - panel.getBoundingClientRect().top;

                // Add visual feedback
                panel.style.cursor = 'grabbing';
                panel.style.opacity = '0.9';

                e.preventDefault();
            });

            document.addEventListener('mousemove', (e) => {
                if (isDragging) {
                    const newLeft = e.clientX - dragOffset.x;
                    const newTop = e.clientY - dragOffset.y;

                    // Apply new position
                    panel.style.left = newLeft + 'px';
                    panel.style.top = newTop + 'px';
                    panel.style.right = 'auto'; // Remove right positioning

                    // Keep panel within viewport bounds
                    const panelRect = panel.getBoundingClientRect();
                    const viewportWidth = window.innerWidth;
                    const viewportHeight = window.innerHeight;

                    if (newLeft < 0) {
                        panel.style.left = '0px';
                    } else if (newLeft + panelRect.width > viewportWidth) {
                        panel.style.left = (viewportWidth - panelRect.width) + 'px';
                    }

                    if (newTop < 0) {
                        panel.style.top = '0px';
                    } else if (newTop + panelRect.height > viewportHeight) {
                        panel.style.top = (viewportHeight - panelRect.height) + 'px';
                    }
                }
            });

            document.addEventListener('mouseup', () => {
                if (isDragging) {
                    isDragging = false;
                    // Restore cursor and opacity
                    panel.style.cursor = 'move';
                    panel.style.opacity = '';

                    // Save position to storage
                    this.savePanelPosition();
                }
            });

            // Load saved position
            this.loadPanelPosition();
        }

        // Add these helper methods to save/load panel position
        savePanelPosition() {
            const panel = document.getElementById('magi-control-panel');
            const rect = panel.getBoundingClientRect();

            GM_setValue('control_panel_position', {
                left: rect.left,
                top: rect.top
            });
        }

        loadPanelPosition() {
            const panel = document.getElementById('magi-control-panel');
            const savedPosition = GM_getValue('control_panel_position');

            if (savedPosition) {
                panel.style.left = savedPosition.left + 'px';
                panel.style.top = savedPosition.top + 'px';
                panel.style.right = 'auto';
            }
        }

        setupMinimizeFunctionality() {
            const panel = document.getElementById('magi-control-panel');
            const minimizeBtn = panel.querySelector('.minimize-btn');

            minimizeBtn.addEventListener('click', (e) => {
                e.stopPropagation();
                this.toggleMinimize();
            });

            // Load saved minimize state
            const isMinimized = GM_getValue('control_panel_minimized', false);
            if (isMinimized) {
                this.minimizePanel();
            }
        }

        toggleMinimize() {
            const panel = document.getElementById('magi-control-panel');
            if (panel.classList.contains('minimized')) {
                this.unminimizePanel();
            } else {
                this.minimizePanel();
            }
        }

        minimizePanel() {
            const panel = document.getElementById('magi-control-panel');

            // Store current position before minimizing
            const currentLeft = panel.style.left;
            const currentTop = panel.style.top;

            panel.classList.add('minimized');
            GM_setValue('control_panel_minimized', true);

            // Restore position after minimizing (in case it was changed by CSS)
            if (currentLeft) panel.style.left = currentLeft;
            if (currentTop) panel.style.top = currentTop;

            // Update tooltip
            const minimizeBtn = panel.querySelector('.minimize-btn');
            minimizeBtn.title = 'Restore panel';
        }

        unminimizePanel() {
            const panel = document.getElementById('magi-control-panel');

            // Store current position before unminimizing
            const currentLeft = panel.style.left;
            const currentTop = panel.style.top;

            panel.classList.remove('minimized');
            GM_setValue('control_panel_minimized', false);

            // Restore position after unminimizing
            if (currentLeft) panel.style.left = currentLeft;
            if (currentTop) panel.style.top = currentTop;

            // Update tooltip
            const minimizeBtn = panel.querySelector('.minimize-btn');
            minimizeBtn.title = 'Minimize panel';
        }

            } else {
                this.disableOCR();
            }
        }

        async enableOCR() {
            this.setupObservers();
            this.processExistingImages();
        }

        disableOCR() {
            this.removeAllOverlays();
            this.processedImages = new WeakSet();
            if (this.observer) this.observer.disconnect();
            if (this.lazyObserver) this.lazyObserver.disconnect();
            if (this.imgLoadListener) document.removeEventListener('load', this.imgLoadListener, true);
            this.imageProcessor.cleanup();
        }

        setupObservers() {
        if (this.observer) this.observer.disconnect();
        if (this.lazyObserver) this.lazyObserver.disconnect();
        if (this.imgLoadListener) document.removeEventListener('load', this.imgLoadListener, true);

        const ip = this.imageProcessor;

        // Lazy observer: visible images jump the queue (v2.1 had `for (const entry of entry)` here,
        // which threw on every callback, so this never worked)
        this.lazyObserver = new IntersectionObserver((entries) => {
            for (const entry of entries) {
                if (entry.isIntersecting) {
                    ip.scheduleProcessing(entry.target, 'high');
                }
            }
        }, {
            rootMargin: APP_CONFIG.PROCESSING.LAZY_LOAD_MARGIN,
            threshold: 0
        });

        const consider = (img) => {
            this.lazyObserver.observe(img);
            this.pageTracker.markDirty();
            ip.scheduleProcessing(img);
        };

        this.observer = new MutationObserver((mutations) => {
            for (const mutation of mutations) {
                for (const node of mutation.addedNodes) {
                    if (node.nodeType === Node.ELEMENT_NODE) {
                        if (node.nodeName === 'IMG') {
                            consider(node);
                        }
                        const images = node.querySelectorAll ? node.querySelectorAll('img') : [];
                        for (const img of images) {
                            consider(img);
                        }
                    }
                }
            }
        });

        this.observer.observe(document.body, {
            childList: true,
            subtree: true
        });

        // Images that finish loading later (lazy loaders, slow connections): queue them the moment
        // they are ready instead of waiting for a timer. 'load' does not bubble, so use capture.
        this.imgLoadListener = (e) => {
            const t = e.target;
            if (t && t.nodeName === 'IMG') {
                this.pageTracker.markDirty();
                ip.scheduleProcessing(t, 'high');
            }
        };
        document.addEventListener('load', this.imgLoadListener, true);

        document.querySelectorAll('img').forEach(img => {
            this.lazyObserver.observe(img);
        });
    }


        setupCleanup() {
            this.cleanupInterval = setInterval(() => {
                imageDataCache.cleanup();
            }, APP_CONFIG.CACHE.CLEANUP_INTERVAL);

            window.addEventListener('beforeunload', () => {
                this.destroy();
            });
        }

        async processAllImages() {
            await this.imageProcessor.processAllImagesWithPriority();
        }

        processExistingImages() {
            if (!this.active) return;

            const images = document.querySelectorAll('img');
            images.forEach(img => {
                if (!img.hasAttribute('data-magi-processed') && img.complete && img.naturalHeight > 0) {
                    this.imageProcessor.scheduleProcessing(img);
                }
            });
        }

        removeAllOverlays() {
            document.querySelectorAll('.manga-bubble').forEach(el => el.remove());
            document.querySelectorAll('[data-magi-processed]').forEach(img => {
                img.removeAttribute('data-magi-processed');
            });
            this.processedImages = new WeakSet();
        }

        forceRepositionBubbles() {
            positioningSystem.forceRepositionAll();
        }

        destroy() {
            this.disableOCR();
            positioningSystem.destroy();
            connectionMonitor.stopPeriodicChecks();

            if (this.googleTranslateObserver) {
                this.googleTranslateObserver.disconnect();
            }

            if (this.cleanupInterval) {
                clearInterval(this.cleanupInterval);
            }

            const panel = document.getElementById('magi-control-panel');
            if (panel) panel.remove();
        }
    }

            const app = new MangaOCRApp();
            app.init();
        }, 0);
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', bootMangaOCR, { once: true });
    } else {
        bootMangaOCR();
    }

    // Export for debugging
    window.MangaOCRApp = MangaOCRApp;
    window.TranslationManager = TranslationManager;
    window.BulkProcessingManager = BulkProcessingManager;
    window.ImageCaptureManager = ImageCaptureManager;
    window.OverlayManager = OverlayManager;
})();
